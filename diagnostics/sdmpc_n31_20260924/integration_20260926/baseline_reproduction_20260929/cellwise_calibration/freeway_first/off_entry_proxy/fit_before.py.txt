"""Finite joint calibration with wider NC states, using the sole existing plant.

The159 cellwise values remain heterogeneous. Twelve coupled update directions
from the existing training Jacobian limit this incremental calculation; this is
not a new complete159-dimensional search or an identifiability claim.
"""
from pathlib import Path
import copy,gzip,json,sys,time
import numpy as np

CW=Path(__file__).resolve().parent
ROOT=next(p for p in CW.parents if (p/'evaluation/controllers/lane_plant_runtime.py').is_file())
sys.path.insert(0,str(ROOT))
from diagnostics.sdmpc_n31_20260924.integration_20260926 import replay_congested_component as r
from evaluation.controllers import lane_plant_runtime as lpr
from diagnostics.demand_sweep.user_native_20260914.metanet_calibration_v1.boundary_factory import ObservationData


def read_prediction(p):
    with gzip.open(p,'rt',encoding='utf-8') as f:return json.load(f)


def main():
    from scipy.optimize import least_squares
    out=CW/'expanded_joint';out.mkdir(exist_ok=False)
    selected=r.load(CW/'jacobian_step/selection.json');spec=r.load(CW/'parameter_spec.json')
    cap=r.load(CW/'expanded_nc_windows/capture.json');catalog=r.load(CW/'data_catalog.json')['checked_records']
    train=[x for x in catalog if x['role']=='train']+[x for x in cap['records'] if x['seed']==29]
    checks=[x for x in catalog if x['case']=='s43_early']+[x for x in cap['records'] if x['seed']==43]
    assert len(train)==13 and len(checks)==6
    protocol=r.load(CW/'protocol.json');document=r.load(spec['initial_manifest'])
    context=lpr.load_sources(spec['initial_manifest']);params=context['parameters']
    original=r.load(ROOT/document['sources']['reference_config']['path'])
    inputs={(x['case'],x['arm']):r.read_primitive_capture(x['input'],x['sha256']) for x in train+checks}
    assert all(a[2]==params for a,k in inputs.values())
    observations={x['truth']:ObservationData(x['truth']) for x in train+checks}
    truth={(x['case'],x['arm']):observations[x['truth']] for x in train}
    initial=np.array([x['initial'] for x in spec['variables']]);z0=np.array(selected['relative_values'])
    lower=np.array([x['lower'] for x in spec['variables']])/initial-1
    upper=np.array([x['upper'] for x in spec['variables']])/initial-1
    jacfile=CW/'jacobian_step/jacobian.npz';old=np.load(jacfile,allow_pickle=False)['jacobian']
    # Exclude the159 prior rows; regularization must not masquerade as data evidence.
    _,singular,vt=np.linalg.svd(old[:-159],full_matrices=False)
    raw=vt[:12].T
    # Every y in[-1,1]^12 respects physical coefficient bounds and +/-15%
    # of the original per-coefficient scale around the frozen fitted point.
    radius=np.maximum(0.,np.minimum(.15,.95*np.minimum(z0-lower,upper-z0)))
    basis=raw*np.minimum(1.,radius/np.maximum(np.sum(abs(raw),axis=1),1e-14))[:,None]
    assert np.all(z0-np.sum(abs(basis),axis=1)>=lower-1e-12)
    assert np.all(z0+np.sum(abs(basis),axis=1)<=upper+1e-12)
    pins={str(ROOT/p):r.sha(ROOT/p) for p in protocol['core_pins']}
    for p in (Path(__file__),Path(r.__file__),jacfile,CW/'parameter_spec.json',CW/'jacobian_step/selection.json',
              CW/'expanded_nc_windows/capture.json',CW/'expanded_nc_windows/summary.json'):
        pins[str(p)]=r.sha(p)
    for x in train+checks:
        pins[x['input']]=x['sha256']
        for name in ('cells_30s.csv','flows_30s.csv','ports_30s.csv','boundaries_30s.csv','geometry.json'):
            p=Path(x['truth'])/name;pins[str(p)]=r.sha(p)
    for p,h in pins.items():assert r.sha(p)==h
    r.save(out/'protocol.json',dict(train=[(x['case'],x['arm']) for x in train],checks=[(x['case'],x['arm']) for x in checks],
        pins=pins,update_directions=12,physical_coefficients=159,coefficient_changes='Existing percell rho/a/tau/nuGE/nuLT and4merge coefficients only.',
        direction_source='Jacobian of previous eight seed29 conditions, prior rows removed. Directions frozen before expanded training.',
        start='Previously frozen trust_0.15',new_states='Five NC seed29 states; old paired response labels retained unchanged.',
        trust_relative_radius=.15,absolute_difference_step_in_direction_coordinate=.1,
        optimizer='least_squares trf; explicit absolute-step Jacobian; max_nfev3, at most39unique evaluations including reusedbaseline.',
        max_new_training_forecasts=494,max_new_baseline_parity=2,max_new_check_forecasts=6,
        loss='Existing state+2*paired response+0.1*relative-value prior. Controller Omega objective unchanged.',
        eligibility='Train objective improves; paired loss no worse; meaningful signs preserved; meanN/q<=110% start. Freeze selection before six checks.',
        limit='Twelve coupled directions, not complete independent159 re-identification. No capacity/reward/control-specific bonus. No future realized boundary.',
        native=0,production_adopted=False,blind_check=False))
    np.savez_compressed(out/'basis.npz',basis=basis,start_relative=z0,singular_values=singular)
    baseline_predictions={};baseline_rows=[]
    saved={ (x['case'],x['arm']):x for x in r.load(CW/'jacobian_step'/selected['selected']/'result.json')['rows'] }
    saved.update({(x['case'],x['arm']):x for x in r.load(CW/'expanded_nc_windows/summary.json')['rows']['cellwise']})
    for x in train:
        key=(x['case'],x['arm']);name=x['case']+'_'+x['arm']
        p=(CW/'jacobian_step'/selected['selected']/(name+'_prediction.json.gz') if x in train[:8]
           else CW/'expanded_nc_windows/cellwise'/(x['case']+'.json.gz'))
        baseline_predictions[key]=read_prediction(p);baseline_rows.append(saved[key]);pins[str(p)]=r.sha(p)
    def model_at(z,folder):
        values=initial*(1+z);config=r._cellwise_config(spec,values,original,params['by_direction']['FW_E']['rho_crit_multiplier'])
        path=folder/'reference_config.json';r.save(path,config)
        manifest=copy.deepcopy(document);manifest['sources']['reference_config']=dict(path=path.relative_to(ROOT).as_posix(),sha256=r.sha(path))
        manifest['qualification']='Expanded joint calibration candidate only; no production/native/SDMPC qualification.'
        path=folder/'manifest.json';r.save(path,manifest)
        return lpr.load_sources(path)['component']
    parity_dir=out/'parity';parity_dir.mkdir();model=model_at(z0,parity_dir);parity={}
    for key in [('s29_early','none'),('s29_t7200','none')]:
        a,k=copy.deepcopy(inputs[key]);pred=json.loads(json.dumps(model.rollout(*a,**k),allow_nan=False))
        parity[str(key)]={f:pred[f]==baseline_predictions[key][f] for f in ('cells','flows','ports','ramps')}
        assert all(parity[str(key)].values())
    r.save(out/'parity.json',dict(exact=parity,new_rollouts=2))
    baseline_loss=r._cellwise_losses(baseline_rows,z0,protocol)
    baseline_residual=r._cellwise_residuals(train,baseline_predictions,truth,baseline_rows,z0,protocol)
    begin=time.perf_counter();cache={};results=[];count=0;best_predictions=baseline_predictions;best_objective=baseline_loss['response_objective']
    def evaluate(y):
        nonlocal count,best_predictions,best_objective
        key=tuple(float(x) for x in y)
        if key in cache:return cache[key]
        assert len(results)<39, 'Finite evaluation budget reached'
        z=z0+basis@np.array(y);folder=out/f'eval_{len(results):03d}';folder.mkdir()
        model=model_at(z,folder);rows=[];predictions={};new=0
        if not results:
            assert np.array_equal(y,np.zeros(12));rows=baseline_rows;predictions=baseline_predictions;residual=baseline_residual
        else:
            for x in train:
                case=(x['case'],x['arm']);a,k=copy.deepcopy(inputs[case]);pred=model.rollout(*a,**k)
                count+=1;new+=1;predictions[case]=pred
                rows.append(r._cellwise_measure(model,pred,x,truth[case]))
            residual=r._cellwise_residuals(train,predictions,truth,rows,z,protocol)
        loss=r._cellwise_losses(rows,z,protocol)
        record=dict(index=len(results),coordinates=list(y),relative_values=z.tolist(),values=(initial*(1+z)).tolist(),rows=rows,loss=loss,new_rollouts=new)
        r.save(folder/'result.json',record);np.save(folder/'residual.npy',residual,allow_pickle=False)
        cache[key]=residual;results.append(record)
        if loss['response_objective']<best_objective:
            best_objective=loss['response_objective'];best_predictions=predictions
        r.save(out/'status.json',dict(stage='calibrating',evaluations=len(results),new_training450=count,
            best_objective=best_objective,elapsed_sec=time.perf_counter()-begin))
        print(json.dumps(dict(evaluation=len(results),new_rollouts=count,objective=loss['response_objective'],response=loss['response'])),flush=True)
        return residual
    def jacobian(y):
        base=evaluate(y);columns=[]
        for j in range(12):
            h=.1 if y[j]+.1<=1 else -.1;trial=np.array(y,copy=True);trial[j]+=h
            columns.append((evaluate(trial)-base)/h)
        return np.column_stack(columns)
    optimize=least_squares(evaluate,np.zeros(12),jac=jacobian,bounds=(-np.ones(12),np.ones(12)),
        method='trf',max_nfev=3,ftol=1e-5,xtol=1e-5,gtol=1e-5)
    chosen=min(results,key=lambda x:x['loss']['response_objective']);loss=chosen['loss']
    gate=loss['response_objective']<baseline_loss['response_objective'] and loss['response']<=baseline_loss['response'] and loss['meaningful_signs_pass'] and loss['mean_n_rmse']<=1.1*baseline_loss['mean_n_rmse'] and loss['mean_q_rmse']<=1.1*baseline_loss['mean_q_rmse']
    r.save(out/'selection.json',dict(selected_index=chosen['index'],eligible=gate,selected=chosen,baseline=baseline_loss,
        frozen_before_checks=True,optimizer=dict(success=bool(optimize.success),nfev=optimize.nfev,njev=optimize.njev,message=str(optimize.message))))
    folder=out/'selected_predictions';folder.mkdir()
    for key,pred in best_predictions.items():
        with gzip.open(folder/('_'.join(key)+'.json.gz'),'wt',encoding='utf-8') as f:json.dump(pred,f,allow_nan=False)
    check=[]
    if gate:
        folder=out/'checks';folder.mkdir();model=lpr.load_sources(out/f"eval_{chosen['index']:03d}/manifest.json")['component']
        for x in checks:
            key=(x['case'],x['arm']);a,k=copy.deepcopy(inputs[key]);pred=model.rollout(*a,**k)
            check.append(r._cellwise_measure(model,pred,x,observations[x['truth']]))
            with gzip.open(folder/('_'.join(key)+'.json.gz'),'wt',encoding='utf-8') as f:json.dump(pred,f,allow_nan=False)
            r.save(out/'status.json',dict(stage='checks',completed_checks=len(check),new_training450=count))
    for p,h in pins.items():assert r.sha(p)==h
    r.save(out/'final_source_pins.json',pins)
    r.save(out/'summary.json',dict(selected_index=chosen['index'],baseline=baseline_loss,selected_loss=loss,training_passed=gate,
        checks=check,check_loss=r._cellwise_losses(check,chosen['relative_values'],protocol) if check else None,
        new_training450=count,new_parity450=2,new_check450=len(check),wall_sec=time.perf_counter()-begin,
        optimizer=dict(success=bool(optimize.success),nfev=optimize.nfev,njev=optimize.njev,message=str(optimize.message)),
        evaluations=len(results),fit=True,production_adopted=False,gain_qualified=False,new_native=0,new_push=0))
    r.save(out/'status.json',dict(stage='complete',new_training450=count,new_parity450=2,new_check450=len(check),production_adopted=False))


def freeway_residuals(records, predictions, observations, measured, factors):
    """Freeway calibration only: states and paired spatial/temporal responses."""
    residual=[]; state_end=0; arrays={};pairs=[]
    def add(values, scale, weight):
        values=np.asarray(values,dtype=float).ravel()
        assert len(values) and np.isfinite(values).all()
        residual.extend(values*np.sqrt(weight/len(values))/scale)
    for x in records:
        key=x['case'],x['arm'];p=predictions[key];o=observations[x['truth']]
        times=[round(x['cutoff']+30*i,6) for i in range(1,16)]
        pc={(round(v['time_s'],6),v['cell']):v for v in p['cells']}
        pf={(round(v['window_end_s'],6),v['cell']):v for v in p['flows']}
        oc={(t,v['cell']):v for t in times for v in o.cells[t] if v['road']=='FW_E'}
        pair={}
        for label in ('actual','predicted'):
            counts=np.array([[float((oc if label=='actual' else pc)[t,c]['n_veh']) for c in range(31)] for t in times])
            flows=[[o.flows[t,'FW_E',c] if label=='actual' else pf[t,c] for c in range(31)] for t in times]
            down=np.array([[float(v['downstream_crossings']) for v in row] for row in flows])
            off=np.array([[float(v['off_departures']) for v in row] for row in flows])
            terminal=np.array([[float(v['terminal_exits_inferred' if label=='actual' else 'terminal_exits']) for v in row] for row in flows])
            pair[label]=dict(n=counts,cumulative_down=np.cumsum(down,axis=0),
                cumulative_exit=np.cumsum(np.column_stack([off[:,c] for c in (9,11,18,20)]+[terminal[:,30]]),axis=0),q=120*(down+off+terminal))
        arrays[key]=pair
        speed=[pc[t,c]['v_kmh']-oc[t,c]['v_kmh'] for t in times for c in range(31)
               if oc[t,c]['n_veh']>=5 and oc[t,c]['v_kmh'] is not None]
        add(speed,20.,.05/len(records))
        add(pair['predicted']['n']-pair['actual']['n'],10.,.10/len(records))
        add(pair['predicted']['q']-pair['actual']['q'],1000.,.10/len(records))
    state_end=len(residual)
    grouped={}
    for x in records:grouped.setdefault(x['case'],[]).append(x)
    n_pairs=sum(len(g)-1 for g in grouped.values())
    for case,group in grouped.items():
        base=next(x for x in group if x['arm'] in ('none','hold'));bkey=case,base['arm']
        for x in group:
            if x is base:continue
            key=case,x['arm'];metrics={}
            for field,floor in (('n',3.),('cumulative_down',10.),('cumulative_exit',5.)):
                actual=arrays[key]['actual'][field]-arrays[bkey]['actual'][field]
                predicted=arrays[key]['predicted'][field]-arrays[bkey]['predicted'][field]
                scale=max(floor,float(np.sqrt(np.mean(actual**2))))
                add(predicted-actual,scale,.5/n_pairs)
                metrics[field]=dict(rmse=float(np.sqrt(np.mean((predicted-actual)**2))),scale=scale)
            rows={m['arm']:m for m in measured if m['case']==case}
            ad=rows[x['arm']]['actual']['mainline_ttt']-rows[base['arm']]['actual']['mainline_ttt']
            pd=rows[x['arm']]['predicted']['mainline_ttt']-rows[base['arm']]['predicted']['mainline_ttt']
            add([pd-ad],max(.5,abs(ad)),.5/n_pairs)
            pairs.append(dict(case=case,arm=x['arm'],actual_delta_mainline=ad,predicted_delta_mainline=pd,metrics=metrics))
    response_end=len(residual)
    add(np.asarray(factors)-1.,1.,.02)
    residual=np.asarray(residual)
    result=dict(objective=float(residual@residual),state=float(residual[:state_end]@residual[:state_end]),
        response=float(residual[state_end:response_end]@residual[state_end:response_end]),pairs=pairs,
        mean_n_rmse=float(np.mean([x['score']['cell_n']['rmse'] for x in measured])),
        mean_q_rmse=float(np.mean([x['score']['flow_vph']['rmse'] for x in measured])))
    return residual,result


def main_freeway():
    """One finite six-factor calibration; no urban/ramp cost in the fit loss."""
    from scipy.optimize import least_squares
    out=CW/'freeway_first/calibration';out.mkdir(exist_ok=False)
    spec=r.load(CW/'parameter_spec.json');old=r.load(CW/'expanded_joint/selection.json')
    values0=np.array(old['selected']['values']);document=r.load(CW/'expanded_joint/eval_036/manifest.json')
    context=lpr.load_sources(CW/'expanded_joint/eval_036/manifest.json');params=context['parameters']
    original=r.load(ROOT/document['sources']['reference_config']['path'])
    catalog=r.load(CW/'data_catalog.json')['checked_records'];cap=r.load(CW/'expanded_nc_windows/capture.json')
    train=[x for x in catalog if x['role']=='train']+[x for x in cap['records'] if x['seed']==29]
    checks=[x for x in catalog if x['case']=='s43_early']+[x for x in cap['records'] if x['seed']==43]+r.load(CW/'additional_admission_s53.json')['records']
    assert len(train)==13 and len(checks)==10
    groups=[('merge21','delta_merge',[21]),('pressure_ge18_22','nu_ge',list(range(18,23))),
            ('pressure_lt18_22','nu_lt',list(range(18,23))),('tau18_22','tau_sec',list(range(18,23))),
            ('tau24_25','tau_sec',[24,25]),('pressure_lt24_25','nu_lt',[24,25])]
    indices=[[i for i,v in enumerate(spec['variables']) if v['parameter']==param and v['cell'] in cells] for _,param,cells in groups]
    assert [len(x) for x in indices]==[1,5,5,5,2,2]
    lower=np.array([max(.5 if j==0 else .75,max(spec['variables'][i]['lower']/values0[i] for i in ix)) for j,ix in enumerate(indices)])
    upper=np.array([min(2. if j==0 else 1.25,min(spec['variables'][i]['upper']/values0[i] for i in ix)) for j,ix in enumerate(indices)])
    assert np.all(lower<=1+1e-10) and np.all(upper>=1-1e-10) and np.all(upper>lower)
    inputs={(x['case'],x['arm']):r.read_primitive_capture(x['input'],x['sha256']) for x in train+checks}
    assert all(a[2]==params for a,k in inputs.values())
    observations={x['truth']:ObservationData(x['truth']) for x in train+checks}
    pins={str(ROOT/p):r.sha(ROOT/p) for p in r.load(CW/'protocol.json')['core_pins']}
    for p in (Path(__file__),Path(r.__file__),CW/'parameter_spec.json',CW/'expanded_joint/selection.json',CW/'additional_admission_s53.json'):
        pins[str(p)]=r.sha(p)
    for x in train+checks:
        pins[x['input']]=x['sha256']
        for name in ('cells_30s.csv','flows_30s.csv','ports_30s.csv','boundaries_30s.csv','geometry.json'):
            p=Path(x['truth'])/name;pins[str(p)]=r.sha(p)
    r.save(out/'protocol.json',dict(groups=[dict(name=g[0],parameter=g[1],cells=g[2],lower=lower[j],upper=upper[j]) for j,g in enumerate(groups)],
        train=[(x['case'],x['arm']) for x in train],checks=[(x['case'],x['arm']) for x in checks],pins=pins,
        loss='Freeway only: state speed/N/q .05/.10/.10; paired N/cumulative-down/cumulative-exit/mainlineTTT .5 each; factor prior .02. Per-pair RMS scales with fixed floors. No ramp/urban cost in fit; still recorded.',
        optimizer='least_squares trf, max_nfev3; absolute factor probes0.05; max24 unique evaluations/299 new training450 plus2parity and10checks.',
        acceptance='At least5% training response improvement, total loss decreases, meanN/q <=110% baseline, mainline signs preserved when actual absolute delta>=.5veh.h.',
        controller_objective_unchanged=True,new_native=0,production_adopted=False,previously_inspected_checks=True))
    predictions0={};rows0=[]
    for x in train:
        key=x['case'],x['arm'];p=CW/'expanded_joint/selected_predictions'/('_'.join(key)+'.json.gz')
        predictions0[key]=read_prediction(p);pins[str(p)]=r.sha(p)
        rows0.append(r._cellwise_measure(context['component'],predictions0[key],x,observations[x['truth']]))
    def model_at(factors,folder):
        values=values0.copy()
        for factor,ix in zip(factors,indices):values[ix]*=factor
        config=r._cellwise_config(spec,values,original,params['by_direction']['FW_E']['rho_crit_multiplier'])
        if np.array_equal(factors,np.ones(6)):assert config==original
        p=folder/'reference_config.json';r.save(p,config)
        doc=copy.deepcopy(document);doc['sources']['reference_config']=dict(path=p.relative_to(ROOT).as_posix(),sha256=r.sha(p))
        doc['qualification']='Freeway response calibration candidate; not full-network/SDMPC/native qualified.'
        p=folder/'manifest.json';r.save(p,doc)
        return lpr.load_sources(p)['component']
    folder=out/'parity';folder.mkdir();model=model_at(np.ones(6),folder);parity=[]
    for key in [('s29_early','rm'),('s29_late','release')]:
        a,k=copy.deepcopy(inputs[key]);pred=json.loads(json.dumps(model.rollout(*a,**k),allow_nan=False))
        assert all(pred[f]==predictions0[key][f] for f in ('cells','flows','ports','ramps'))
        parity.append(key)
    r.save(out/'parity.json',dict(exact=parity,new_rollouts=2))
    base_res,base_loss=freeway_residuals(train,predictions0,observations,rows0,np.ones(6))
    start=time.perf_counter();cache={};results=[];best_predictions=predictions0;best=base_loss['objective'];count=0
    def evaluate(factors):
        nonlocal best,best_predictions,count
        key=tuple(float(x) for x in factors)
        if key in cache:return cache[key]
        assert len(results)<24
        folder=out/f'eval_{len(results):03d}';folder.mkdir();model=model_at(factors,folder)
        if np.array_equal(factors,np.ones(6)):
            rows=rows0;predictions=predictions0;res,loss=base_res,base_loss;new=0
        else:
            rows=[];predictions={};new=0
            for x in train:
                name=x['case'],x['arm'];a,k=copy.deepcopy(inputs[name]);pred=model.rollout(*a,**k)
                count+=1;new+=1;predictions[name]=pred;rows.append(r._cellwise_measure(model,pred,x,observations[x['truth']]))
            res,loss=freeway_residuals(train,predictions,observations,rows,factors)
        record=dict(index=len(results),factors=list(factors),loss=loss,rows=rows,new_rollouts=new)
        r.save(folder/'result.json',record);results.append(record);cache[key]=res
        if loss['objective']<best:best=loss['objective'];best_predictions=predictions
        r.save(out/'status.json',dict(stage='calibrating',evaluations=len(results),new_training450=count,elapsed_sec=time.perf_counter()-start))
        print(json.dumps(dict(evaluation=len(results),new_training450=count,loss=loss['objective'],response=loss['response'])),flush=True)
        return res
    evaluate(np.ones(6))
    def jac(factors):
        base=evaluate(factors);cols=[]
        for j in range(6):
            up=upper[j]-factors[j];down=factors[j]-lower[j];h=min(.05,up) if up>=down else -min(.05,down)
            assert abs(h)>1e-9
            trial=np.array(factors,copy=True);trial[j]+=h;cols.append((evaluate(trial)-base)/h)
        return np.column_stack(cols)
    fit=least_squares(evaluate,np.ones(6),jac=jac,bounds=(lower,upper),method='trf',max_nfev=3)
    chosen=min(results,key=lambda x:x['loss']['objective']);loss=chosen['loss']
    gate=(loss['objective']<base_loss['objective'] and loss['response']<=.95*base_loss['response']
          and loss['mean_n_rmse']<=1.1*base_loss['mean_n_rmse'] and loss['mean_q_rmse']<=1.1*base_loss['mean_q_rmse']
          and all(x['actual_delta_mainline']*x['predicted_delta_mainline']>0 for x in loss['pairs'] if abs(x['actual_delta_mainline'])>=.5))
    r.save(out/'selection.json',dict(selected_index=chosen['index'],selected=chosen,baseline=base_loss,eligible=gate,frozen_before_checks=True))
    folder=out/'selected_predictions';folder.mkdir()
    for key,pred in best_predictions.items():
        with gzip.open(folder/('_'.join(key)+'.json.gz'),'wt',encoding='utf8') as f:json.dump(pred,f,allow_nan=False)
    check_rows=[];check_predictions={}
    if gate:
        folder=out/'checks';folder.mkdir();model=lpr.load_sources(out/f"eval_{chosen['index']:03d}/manifest.json")['component']
        for x in checks:
            key=x['case'],x['arm'];a,k=copy.deepcopy(inputs[key]);pred=model.rollout(*a,**k);check_predictions[key]=pred
            check_rows.append(r._cellwise_measure(model,pred,x,observations[x['truth']]))
            with gzip.open(folder/('_'.join(key)+'.json.gz'),'wt',encoding='utf8') as f:json.dump(pred,f,allow_nan=False)
    check_loss=freeway_residuals(checks,check_predictions,observations,check_rows,chosen['factors'])[1] if check_rows else None
    for p,h in pins.items():assert r.sha(p)==h,p
    r.save(out/'final_source_pins.json',pins)
    r.save(out/'summary.json',dict(selected_index=chosen['index'],baseline=base_loss,selected_loss=loss,training_passed=gate,
        checks=check_rows,check_loss=check_loss,new_training450=count,new_parity450=2,new_check450=len(check_rows),
        wall_sec=time.perf_counter()-start,evaluations=len(results),optimizer=dict(success=bool(fit.success),nfev=fit.nfev,njev=fit.njev,message=str(fit.message)),
        new_native=0,new_push=0,production_adopted=False,gain_qualified=False))
    r.save(out/'status.json',dict(stage='complete',training_passed=gate,new_training450=count,new_check450=len(check_rows)))


def main_local_fd():
    """Three predeclared local FD candidates, frozen before the ten checks."""
    out=CW/'freeway_first/local_fd24_25';out.mkdir(exist_ok=False)
    manifest=CW/'expanded_joint/eval_036/manifest.json';document=r.load(manifest)
    context=lpr.load_sources(manifest);params=context['parameters'];model0=context['component']
    original=r.load(ROOT/document['sources']['reference_config']['path'])
    spec=r.load(CW/'parameter_spec.json');values0=np.array(r.load(CW/'expanded_joint/selection.json')['selected']['values'])
    cat=r.load(CW/'data_catalog.json')['checked_records'];cap=r.load(CW/'expanded_nc_windows/capture.json')['records']
    train=[x for x in cat if x['role']=='train']+[x for x in cap if x['seed']==29]
    checks=[x for x in cat if x['case']=='s43_early']+[x for x in cap if x['seed']==43]+r.load(CW/'additional_admission_s53.json')['records']
    assert len(train)==13 and len(checks)==10
    plans=[('shape125',1.,1.25),('rho120_shape125',1.2,1.25),('rho120_shape150',1.2,1.5)]
    pins={str(ROOT/p):r.sha(ROOT/p) for p in r.load(CW/'protocol.json')['core_pins']}
    for p in (Path(__file__),Path(r.__file__),manifest,CW/'parameter_spec.json'):pins[str(p)]=r.sha(p)
    inputs={(x['case'],x['arm']):r.read_primitive_capture(x['input'],x['sha256']) for x in train+checks}
    assert all(a[2]==params for a,k in inputs.values())
    observations={x['truth']:ObservationData(x['truth']) for x in train+checks}
    for x in train+checks:
        pins[x['input']]=x['sha256']
        for n in ('cells_30s.csv','flows_30s.csv','ports_30s.csv','boundaries_30s.csv','geometry.json'):
            p=Path(x['truth'])/n;pins[str(p)]=r.sha(p)
    r.save(out/'protocol.json',dict(plans=plans,cells=[24,25],parameters=['rho_crit','metanet_a_m'],
        train=[(x['case'],x['arm']) for x in train],checks=[(x['case'],x['arm']) for x in checks],
        max_training450=39,max_checks450=10,no_refit_on_checks=True,
        acceptance='At least5% freeway response improvement; meanN/q no more than110% baseline; local24/25 speedRMSE improves; meaningful mainline signs correct. Select by training response only.',
        purpose='Local FD desired-speed bias, not an arbitrary recovery/control reward.',
        conditional_diagnostic_informed_structure_but_not_fit=True,blind_checks=False,pins=pins))
    def local_error(records,predictions):
        vv=[]
        for x in records:
            pred=predictions[x['case'],x['arm']]
            truth=observations[x['truth']]
            obs={(t,z['cell']):z for t,rows in truth.cells.items() for z in rows if z['road']=='FW_E'}
            for row in pred['cells']:
                if row['cell'] not in (24,25):continue
                label=obs[round(row['time_s'],6),row['cell']]['v_kmh']
                if label is not None:vv.append((row['v_kmh']-label)**2)
        return float(np.sqrt(np.mean(vv)))
    def evaluate(name,model,records,cached=False):
        rows=[];predictions={}
        folder=out/name;folder.mkdir()
        for x in records:
            key=x['case'],x['arm']
            if cached:
                p=CW/'expanded_joint/selected_predictions'/('_'.join(key)+'.json.gz')
                pins[str(p)]=r.sha(p);pred=read_prediction(p)
            else:
                a,k=copy.deepcopy(inputs[key]);pred=model.rollout(*a,**k)
                with gzip.open(folder/('_'.join(key)+'.json.gz'),'wt',encoding='utf8') as f:json.dump(pred,f,allow_nan=False)
            rows.append(r._cellwise_measure(model,pred,x,observations[x['truth']]));predictions[key]=pred
        loss=freeway_residuals(records,predictions,observations,rows,np.ones(6))[1]
        loss['local_speed_rmse']=local_error(records,predictions)
        result=dict(name=name,loss=loss,rows=rows)
        r.save(folder/'result.json',result)
        print(json.dumps(dict(name=name,response=loss['response'],local_v_rmse=loss['local_speed_rmse'])),flush=True)
        return result
    baseline=evaluate('baseline',model0,train,True);results=[]
    for name,rho_factor,shape_factor in plans:
        folder=out/(name+'_config');folder.mkdir()
        values=values0.copy()
        for idx,v in enumerate(spec['variables']):
            if v['cell'] in (24,25) and v['parameter'] in ('rho_crit','metanet_a_m'):
                values[idx]*=rho_factor if v['parameter']=='rho_crit' else shape_factor
                assert v['lower']<=values[idx]<=v['upper']
        config=r._cellwise_config(spec,values,original,params['by_direction']['FW_E']['rho_crit_multiplier'])
        p=folder/'reference_config.json';r.save(p,config)
        doc=copy.deepcopy(document);doc['sources']['reference_config']=dict(path=p.relative_to(ROOT).as_posix(),sha256=r.sha(p))
        doc['qualification']='Local FD candidate only; no adoption or whole-network qualification.'
        m=folder/'manifest.json';r.save(m,doc);model=lpr.load_sources(m)['component']
        item=evaluate(name,model,train)
        a,b=item['loss'],baseline['loss']
        item['eligible']=(a['response']<=.95*b['response'] and a['mean_n_rmse']<=1.1*b['mean_n_rmse'] and
            a['mean_q_rmse']<=1.1*b['mean_q_rmse'] and a['local_speed_rmse']<b['local_speed_rmse'] and
            all(z['actual_delta_mainline']*z['predicted_delta_mainline']>0 for z in a['pairs'] if abs(z['actual_delta_mainline'])>=.5))
        results.append(item);r.save(out/'partial_results.json',results)
    eligible=[z for z in results if z['eligible']]
    selected=min(eligible,key=lambda z:z['loss']['response'])['name'] if eligible else None
    r.save(out/'selection.json',dict(selected=selected,eligible=[z['name'] for z in eligible],frozen_before_checks=True))
    checked=None
    if selected:
        m=out/(selected+'_config')/'manifest.json'
        checked=evaluate('checks',lpr.load_sources(m)['component'],checks)
    for p,h in pins.items():assert r.sha(p)==h,p
    r.save(out/'summary.json',dict(baseline=baseline,results=results,selected=selected,checks=checked,
        source_pins=pins,new_training450=39,new_check450=10 if checked else 0,
        new_native=0,production_adopted=False,gain_qualified=False))
    r.save(out/'status.json',dict(stage='complete',selected=selected,new_training450=39,new_check450=10 if checked else 0))


if __name__=='__main__':
    if '--local-fd24-25' in sys.argv[1:]:main_local_fd()
    elif '--freeway-first' in sys.argv[1:]:main_freeway()
    else:main()
