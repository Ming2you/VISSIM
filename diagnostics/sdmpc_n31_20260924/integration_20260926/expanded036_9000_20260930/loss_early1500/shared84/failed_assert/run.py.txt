def audited(captured, reference, output, **kwargs):
    from evaluation.controllers import area_leader_objective as constraints
    from src.controllers import rollout_endpoint as endpoint
    from src.controllers.priced_wu_link_controller import LinkAgentWuFollower
    cfg = captured['cfg']
    cfg_bytes = pickle.dumps(cfg, protocol=5)
    quantity_cfg = copy.deepcopy(cfg)
    follower = LinkAgentWuFollower(quantity_cfg)
    assert pickle.dumps(cfg, protocol=5) == cfg_bytes
    assert len(follower.cfg.network.signals) == 17
    evaluate = endpoint.evaluate_price_point
    from evaluation.controllers import lane_ramp_runtime as ramp_runtime
    gap_original = ramp_runtime.gap_acceptance_supply_vph
    gap_rows = []
    shared_enabled = shared_case is not None and shared_case!='recovery_base'
    if shared_enabled:
        from evaluation.controllers import area_freeway_accounting as accounting
        from evaluation.controllers.physical_lane_groups import hadi_receiving_vph
        kernel=captured['state'].lane_freeway_runtime.configs['FW_E']
        net=kernel.network
        assert not getattr(net,'freeway_hadiuzzaman',None)
        assert net.rho_max==180 and net.rho_crit==37.8 and net.freeway_capacity_veh_h==6937
        assert not getattr(net,'vsl_fd_two_branch',False)
        lengths=accounting.cell_lengths_km(kernel,'FW_E',31)
        cells=[]
        for i,length in enumerate(lengths):
            width=net.freeway_segment_lanes['FW_E'][i]
            # Other cells reproduce the existing storage/one-step receiving.
            q=net.freeway_capacity_veh_h/width if i==23 else net.rho_max*length/kernel.simulation.T_f_h
            wave=q/(net.rho_max-net.rho_crit) if i==23 else length/kernel.simulation.T_f_h
            spec=dict(capacity_vphpl=q,wave_kmh=wave,rho_critical=net.rho_crit,theta=0.)
            for rho in (0.,net.rho_crit,60.,179.,180.):
                target=(q*min(1.,max(0.,(net.rho_max-rho)/(net.rho_max-net.rho_crit)))
                        if i==23 else (net.rho_max-rho)*length/kernel.simulation.T_f_h)
                assert abs(hadi_receiving_vph(rho,net.rho_max,net.rho_crit,q,wave,0.)-target)<1e-7
            cells.append(spec)
        # Use only the existing shared receiving allocator. An empty relaxation
        # scope preserves all METANET speed/merge/drop coefficients and targets.
        net.freeway_hadiuzzaman={'FW_E':dict(ctm=True,relaxation='fd_cap',relaxation_cells=[],
            congested_branch='fixed_wave',cells=cells)}
        (O/'runtime_receiving_config.json').write_text(json.dumps(net.freeway_hadiuzzaman,indent=2)+'\n',encoding='utf8')
    def gap_observer(conflict, critical, followup):
        local = sys._getframe(1).f_locals
        result = gap_original(conflict, critical, followup)
        if local['name']=='RM_C10484':
            state, net = local['state'],local['cfg'].network
            road = local['road']
            cell = net.ramp_merge_segment_index[local['name']]
            assert local['mapping'] is None and local['plant'] is None
            assert local['i']==cell-1 and local['split']==0
            assert cell==23 and critical==2 and followup==1.5
            if shared_enabled:
                assert local['demand'].incident_capacity_factor==1
                width=state.freeway_effective_lanes[road][cell-1]
                assert width==state.freeway_effective_lanes[road][cell]==3
                supply=net.freeway_capacity_veh_h*min(1.,max(0.,
                    (net.rho_max-state.freeway_density[road][cell])/(net.rho_max-net.rho_crit)))
                mainline=conflict*width
                result=min(result,max(0.,supply-mainline))
            gap_rows.append(dict(time_sec=local['freeway'].time_sec,
                q_conflict_vph=conflict,gap_supply_vph=result,
                canonical_supply_vph=local['canonical_rate'],
                upstream_cell=cell-1,upstream_density=state.freeway_density[road][cell-1],
                upstream_speed=state.freeway_speed[road][cell-1],
                merge_cell=cell,merge_density=state.freeway_density[road][cell],
                merge_speed=state.freeway_speed[road][cell],
                jam_density=net.rho_max,
                ramp_stock_veh=state.ramp_queue[local['name']],
                lane_resolved_mainline=False,
                **(dict(shared_supply_vph=supply,mainline_request_vph=mainline,
                         old_gap_vph=gap_original(conflict,critical,followup)) if shared_enabled else {})))
        return result
    if receiving_trace or shared_enabled:
        ramp_runtime.gap_acceptance_supply_vph=gap_observer

    def checked(state, action, *args, **kw):
        assert len(rows) < (1 if single else 4)
        point = evaluate(state, action, *args, **kw)
        if not rows and old is not None:
            assert abs(point.ttt-old['ttt_omega_veh_h']) < 1e-9, (point.ttt, old['ttt_omega_veh_h'])
        q = constraints.shared_urban_quantities(follower, point.control_area_response,
            start_sec=state.time_sec, horizon_steps=3)
        budget = constraints.shared_quantity_constraints(follower, action, q,
            start_sec=state.time_sec, horizon_steps=3,
            np_mode=caps['np']['mode'], target_np_veh=caps['np']['target'],
            np_tolerance_veh=caps['np']['tolerance'], nuf_mode=caps['nuf']['mode'],
            target_nuf_veh_h=caps['nuf']['target'], nuf_tolerance_veh_h=caps['nuf']['tolerance'])
        name = list(protocol['conditions'])[len(rows)]
        rows.append(dict(case=name, ttt=point.ttt, resource=budget, quantities=q))
        (O/'quantities.json').write_text(json.dumps(rows,indent=2)+'\n',encoding='utf8')
        return point

    endpoint.evaluate_price_point = checked
    try:
        original(captured, reference, output, **dict(kwargs,
            onset_factorial=not early and not single,
            early_ramp_factorial=early, first_interval_audit=True,
            only_reference=True if single else kwargs.get('only_reference',False)))
    finally:
        endpoint.evaluate_price_point = evaluate
        ramp_runtime.gap_acceptance_supply_vph=gap_original
    assert len(rows) == (1 if single else 4)
    if receiving_trace:
        assert len(gap_rows)==450 and [r['time_sec'] for r in gap_rows]==list(range(1500,1950))
        (O/'receiving_trace.json').write_text(json.dumps(gap_rows,indent=2)+'\n',encoding='utf8')
        saved=json.loads((I/'closedloop_recorded1500_lever450_RM_C10484_trace10484_early1500/held_actual.json').read_bytes())
        current=json.loads((output/'held_actual.json').read_bytes())
        saved.pop('wall_sec');current.pop('wall_sec')
        assert saved==current, 'Instrumentation must preserve all saved forecast values'
    if shared_enabled:
        assert len(gap_rows)==450 and [r['time_sec'] for r in gap_rows]==list(range(at,at+450))
        (O/'receiving_trace.json').write_text(json.dumps(gap_rows,indent=2)+'\n',encoding='utf8')
    (O/'output.json').write_text(json.dumps(dict(output=str(output)),indent=2),encoding='utf8')
