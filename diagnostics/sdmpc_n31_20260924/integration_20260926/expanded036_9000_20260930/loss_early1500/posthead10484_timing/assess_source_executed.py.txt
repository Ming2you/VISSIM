"""Assess the four completed1500s forecasts against the same-command150s prefix."""
import csv
import hashlib
import gzip
import json
from pathlib import Path

O = Path(__file__).resolve().parent
E, I = O.parent, O.parents[1]
P = I/'closedloop_recorded1500_lever450_RM_C10484_trace10484_early1500'
D = Path('D:/VISSIM_runs/20260930_expanded036_s29_9000_r2/sdmpc/decisions_sdmpc31_sdmpc9000_s29')
pins = {}


def read(p):
    b = p.read_bytes()
    pins[str(p)] = hashlib.sha256(b).hexdigest()
    return b.decode('utf-8-sig')


def js(p):
    return json.loads(read(p))


def table(p):
    return list(csv.DictReader(read(p).splitlines()))


def main():
    done = js(O/'run_completion.json')
    assert done['status'] == 'complete' and done['forecasts'] == 4 and done['pins_unchanged']
    result = js(P/'summary.json')['results']
    assert list(result) == ['held_actual','rm10490','rm10484','both_ramps']
    base = result['held_actual']
    quantity = {r['case']: r for r in js(O/'quantities.json')}
    captures = {t: js(D/f'state_{t:06d}.json')['vehicle_records'] for t in (1500,1650)}
    assert all(r['complete'] and r['unobservable_count']==0 for r in captures.values())
    vehicles = {t: r['records'] for t,r in captures.items()}
    detector_rows = [r for r in table(E/'loss_diagnosis/detector_windows_150s.csv')
                     if r['arm']=='sdmpc' and float(r['sim_sec'])==1650]
    locations = table(I/'selected/obs150/obs150_detectors_v2.csv')
    native = {}
    for ramp in base['ramps']:
        con = ramp.removeprefix('RM_C')
        stock = {t: [r for r in vs if str(r['link_no'])==con] for t,vs in vehicles.items()}
        def count(role):
            rows = [r for r in detector_rows if r['role']==role and r['link']==con]
            assert rows
            return sum(int(r['count']) for r in rows)
        def position(role):
            rows = [r for r in locations if r['role']==role and r['link']==con]
            values = {int(r['lane']):float(r['pos']) for r in rows}
            assert values and len(values)==len(rows)
            return values
        arrival_pos, head_pos = position('ramp_arrival'), position('meter_head')
        pre = {t: sum(v['position_m']<arrival_pos[v['lane_no']] for v in vs) for t,vs in stock.items()}
        post = {t: sum(v['position_m']>=head_pos[v['lane_no']] for v in vs) for t,vs in stock.items()}
        arrival = count('ramp_arrival')+pre[1650]-pre[1500]
        head = count('meter_head')
        merge = head+post[1500]-post[1650]
        assert len(stock[1500])+arrival-merge==len(stock[1650])
        native[ramp] = dict(initial_stock=len(stock[1500]),arrival=arrival,head=head,
            merge=merge,final_stock=len(stock[1650]),initial_posthead=post[1500],
            final_posthead=post[1650])
    geometry = js(I/'selected/port_gain/geometry.json')
    addr = geometry['addresses']
    from bisect import bisect_right
    cell_truth = {}
    for t,vs in vehicles.items():
        cell_truth[t] = {}
        for road in ('FW_E','FW_W'):
            cells = sorted([c for c in geometry['cells'] if c['road']==road],key=lambda c:c['cell'])
            ends = [c['end_m'] for c in cells]
            bins = [[] for _ in cells]
            for v in vs:
                item = addr.get(str(v['link_no']))
                if item and item[0]==road:
                    cell = min(bisect_right(ends,item[1]+v['position_m']),30)
                    bins[cell].append(v['speed_kph'])
            cell_truth[t][road] = dict(n=[len(b) for b in bins],
                speed=[sum(b)/len(b) if b else None for b in bins])
    rows = []
    for name,r in result.items():
        first = r['first_interval']
        assert first['physical_cell_states'][0]==base['first_interval']['physical_cell_states'][0]
        for road in ('FW_E','FW_W'):
            model_n = first['physical_cell_states'][0]['vehicle_count'][road]
            assert max(abs(a-b) for a,b in zip(model_n,cell_truth[1500][road]['n']))<1e-8
        targets = ({'RM_C10490'} if name=='rm10490' else {'RM_C10484'} if name=='rm10484'
                   else {'RM_C10490','RM_C10484'} if name=='both_ramps' else set())
        for block,(cmd,held) in enumerate(zip(r['commands'],base['commands'])):
            assert all(cmd[k]==held[k] for k in ('vsl','green_times','offsets'))
            assert all(v==110 for v in cmd['vsl'].values())
            assert all(v==(8-2*block if k in targets else 10) for k,v in cmd['meters'].items())
        delta = {k:r['cost_by_stock'].get(k,0)-base['cost_by_stock'].get(k,0) for k in set(r['cost_by_stock'])|set(base['cost_by_stock'])}
        # Report exact model stock addresses; keep origin/merge stocks separate.
        east = delta.get('freeway:FW_E',0)
        west = delta.get('freeway:FW_W',0)
        ramps = sum(v for k,v in delta.items() if k.startswith('ramp:'))
        total = r['ttt_omega_veh_h']-base['ttt_omega_veh_h']
        assert abs(sum(delta.values())-total)<1e-7
        assert all(abs(v['residual'])<1e-7 for v in r['ramps'].values())
        rows.append(dict(case=name,delta_omega=total,delta_E=east,delta_W=west,delta_ramps=ramps,
            delta_other_omega=total-east-west-ramps,delta_outside=r['tracked_outside_residence_veh_h']-base['tracked_outside_residence_veh_h'],
            delta_merge={k:v['merge']-base['ramps'][k]['merge'] for k,v in r['ramps'].items()},
            first150_delta_merge={k:v['merge']-base['first_interval']['ramps'][k]['merge'] for k,v in first['ramps'].items()},
            n_p=quantity[name]['resource']['np'],n_uf=quantity[name]['resource']['nuf'],
            feasible=quantity[name]['resource']['feasible']))
    actual_first = dict(ramps=native,cells=cell_truth,
        model_ramps=base['first_interval']['ramps'],model_cells=base['first_interval']['physical_cell_states'][1])
    trace_path = P/'held_actual_RM_C10484_trace.json.gz'
    pins[str(trace_path)] = hashlib.sha256(trace_path.read_bytes()).hexdigest()
    with gzip.open(trace_path,'rt',encoding='utf-8') as f:
        trace = json.load(f)
    grouped = {}
    prefix = 'physical_ramp_merge_'
    for r in trace['resources']:
        if r['kind'].startswith(prefix):
            grouped.setdefault(r['start_sec'],{})[r['kind'].removeprefix(prefix)] = r
    gate_blocks = []
    for start in (1500,1650,1800):
        ties = {k:0 for k in ('stock','canonical_receiving','physical_receiving','eligible')}
        total = 0.
        for sec in range(start,start+150):
            gates = grouped[sec]
            assert set(gates)==set(ties)
            cap = min(r['available_veh'] for r in gates.values())
            used = gates['eligible']['accepted_total_veh']
            assert abs(used-cap)<1e-8
            total += used
            for k,r in gates.items():
                ties[k] += abs(r['available_veh']-cap)<1e-8
        gate_blocks.append(dict(start_sec=start,end_sec=start+150,merge=total,binding_seconds_including_ties=ties))
    initial_post = trace['initial_stock']['downstream_travelling_veh']+trace['initial_stock']['merge_ready_veh']
    model_head = base['first_interval']['ramps']['RM_C10484']['head']
    model_merge = base['first_interval']['ramps']['RM_C10484']['merge']
    posthead = dict(native=native['RM_C10484'],model_initial=initial_post,
        model_final=initial_post+model_head-model_merge,model_head=model_head,model_merge=model_merge,
        fixed_posthead_speed_kmh=trace['initial_buffer']['posthead_travel_speed_kmh'],
        gate_blocks=gate_blocks,scope='Recorded model gates; not a native capacity estimate or causal intervention')
    for p,h in js(O/'execution_pins.json').items():
        assert hashlib.sha256(Path(p).read_bytes()).hexdigest()==h
    for p,h in pins.items():
        assert hashlib.sha256(Path(p).read_bytes()).hexdigest()==h
    output = dict(status='complete',rows=rows,first150=actual_first,posthead10484=posthead,all_commands_valid=True,
        all_initial_cells_exact=True,all8_ramp_balances_pass=True,source_pins=pins,
        forecast_compute_seconds=sum(r['wall_sec'] for r in result.values()),run_wall_seconds=done['wall_sec'],
        new_native=0,new_forecasts=0,model_response_only=True,adopted=False,goal_qualified=False)
    (O/'assessment.json').write_text(json.dumps(output,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(dict(rows=rows,native_first150=native,model_first150=base['first_interval']['ramps']),ensure_ascii=False))


def posthead_timing():
    """Isolate future head timing with frozen receiving; not a plant forecast."""
    import ast
    import math
    import statistics

    out = O/'posthead10484_timing'
    out.mkdir(exist_ok=True)
    assert not (out/'verification.json').exists(), 'Preserve completed diagnostic'
    protocol = dict(window=[1500,1650], ramp='RM_C10484',
        cases=['saved_head_parity','native_head_times','native_head_ceil_1s'],
        fixed='Initial cohorts, nominal travel and saved canonical/physical receiving budgets',
        scope='Future-conditioned buffer-only accounting; changed merges do not update freeway',
        full_forecasts=0,fit=0,native_runs=0,fzp_scans=0,live_monitoring=0)
    (out/'protocol.json').write_text(json.dumps(protocol,indent=2)+'\n',encoding='utf-8')
    # Reuse only the two pure functions. Importing the historical script would
    # re-execute its already completed 10490 diagnosis and rewrite its files.
    helper = E/'audit_posthead10490_receiving.py'
    tree = ast.parse(read(helper))
    functions = [n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in ('close','fluid_replay')]
    assert len(functions)==2
    namespace = {}
    exec(compile(ast.Module(body=functions,type_ignores=[]),str(helper),'exec'),namespace)
    replay, close = namespace['fluid_replay'], namespace['close']
    assessment = js(O/'assessment.json')
    for path,digest in assessment['source_pins'].items():
        assert hashlib.sha256(Path(path).read_bytes()).hexdigest()==digest, path
    trace_path = P/'held_actual_RM_C10484_trace.json.gz'
    data = trace_path.read_bytes()
    pins[str(trace_path)] = hashlib.sha256(data).hexdigest()
    trace = json.loads(gzip.decompress(data))
    initial = trace['initial_buffer']
    start, stop = protocol['window']
    states = {t:js(D/f'state_{t:06d}.json') for t in (start,stop)}
    loc = [r for r in table(I/'selected/obs150/obs150_detectors_v2.csv')
           if r['role']=='meter_head' and r['link']=='10484']
    assert len(loc)==1 and int(loc[0]['lane'])==1
    det = int(loc[0]['dcp_no'])
    assert abs(float(loc[0]['pos'])-initial['head_position_m'])<1e-6
    events = []
    for state in states.values():
        spec = state['obs150']['mer']
        path = D/spec['chunk']
        text = read(path)
        assert pins[str(path)]==spec['chunk_sha256']
        events.extend(json.loads(line) for line in text.splitlines())
    assert len({r[0] for r in events})==len(events)
    entries = [r for r in events if r[1]==det and r[2] is not None]
    assert len({r[4] for r in entries})==len(entries)
    native_heads = [(r[2],1.) for r in entries if start<=r[2]<stop]
    close(len(native_heads),states[stop]['obs150']['detectors'][str(det)])
    close(len(native_heads),states[stop]['obs150']['detectors_cum'][str(det)]-
          states[start]['obs150']['detectors_cum'][str(det)])
    native = assessment['posthead10484']['native']
    close(len(native_heads),native['head'])
    travel = initial['post_head_travel_sec']
    due = [(start+(initial['length_m']-p)/initial['posthead_travel_speed_kmh']*3.6,n)
           for p,_,n in initial['initial_cohorts'] if p>=initial['head_position_m']]
    close(sum(n for _,n in due),native['initial_posthead'])
    grouped = {}
    model_heads = []
    for r in trace['resources']:
        if not start<r['end_sec']<=stop:
            continue
        if r['kind']=='physical_ramp_head_service':
            assert r['resource']=='RM_C10484:0'
            model_heads.append((r['end_sec'],r['accepted_total_veh']))
        elif r['kind'] in ('physical_ramp_merge_canonical_receiving','physical_ramp_merge_physical_receiving'):
            assert r['resource']=='RM_C10484'
            grouped.setdefault(r['end_sec'],[]).append(r['available_veh'])
    assert len(grouped)==len(model_heads)==150 and all(len(v)==2 for v in grouped.values())
    budgets = [dict(end_sec=t,available_veh=min(v)) for t,v in sorted(grouped.items())]
    outcomes = {name:replay(due,heads,travel,budgets) for name,heads in (
        ('saved_head_parity',model_heads),('native_head_times',native_heads),
        ('native_head_ceil_1s',[(math.ceil(t),n) for t,n in native_heads]))}
    close(outcomes['saved_head_parity']['merge'],assessment['posthead10484']['model_merge'])
    close(outcomes['saved_head_parity']['end_post'],assessment['posthead10484']['model_final'])
    by_id = {r[4]:r[2] for r in entries}
    post = [v for v in states[stop]['vehicle_records']['records']
            if v['link_no']==10484 and v['position_m']>=initial['head_position_m']]
    close(len(post),native['final_posthead'])
    assert all(v['veh_no'] in by_id for v in post), 'Head ages require complete observed crossings'
    ages = [dict(vehicle=v['veh_no'],head_sec=by_id[v['veh_no']],age_sec=stop-by_id[v['veh_no']],
                 position_m=v['position_m'],speed_kmh=v['speed_kph']) for v in post]
    blocks = [dict(start_sec=t,end_sec=t+30,
        native_head=sum(n for at,n in native_heads if t<=at<t+30),
        model_head=sum(n for at,n in model_heads if t<at<=t+30),
        saved_receiving=sum(r['available_veh'] for r in budgets if t<r['end_sec']<=t+30))
        for t in range(start,stop,30)]
    for path,digest in pins.items():
        assert hashlib.sha256(Path(path).read_bytes()).hexdigest()==digest, path
    result = dict(status='complete_diagnostic_only',protocol=protocol,native=native,
        nominal_posthead_travel_sec=travel,outcomes=outcomes,head_30s=blocks,
        native_end_post_vehicles=ages,native_head_age_median_sec=statistics.median(v['age_sec'] for v in ages),
        native_post_older_than_nominal=sum(v['age_sec']>travel for v in ages),
        native_post_speed_below5=sum(v['speed_kmh']<5 for v in ages),
        parity_pass=True,head_counts_verified=True,source_pins=pins,
        mass_conserved=True,autonomous=False,adopted=False,goal_qualified=False)
    (out/'verification.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    (out/'assess_source_executed.py.txt').write_bytes(Path(__file__).read_bytes())
    print(json.dumps({k:v for k,v in result.items() if k not in ('source_pins','native_end_post_vehicles')},ensure_ascii=False))


if __name__=='__main__':
    import sys
    if sys.argv[1:]==['--posthead-timing']:
        posthead_timing()
    else:
        assert not sys.argv[1:]
        main()
