function Hash([string]$p) { (Get-FileHash -LiteralPath $p -Algorithm SHA256).Hash.ToLowerInvariant() }
function Get-UnreservedVissim($live) {
    # An optional protocol pins already-owned concurrent runs. Never stop them.
    $reserved=@($protocol.allowed_concurrent_vissim | Where-Object { $null -ne $_ })
    if ($reserved.Count -gt 3 -or @($reserved.pid | Select-Object -Unique).Count -ne $reserved.Count) {
        throw 'Invalid concurrent VISSIM ownership list'
    }
    foreach ($process in $live) {
        $match=@($reserved | Where-Object { [int]$_.pid -eq [int]$process.ProcessId })
        if (-not $match.Count) { $process; continue }
        $created=[DateTimeOffset]$match[0].creation_time
        if ($null -eq $process.CreationDate -or
            [Math]::Abs(($process.CreationDate.ToUniversalTime()-$created.UtcDateTime).TotalSeconds) -gt .001) {
            throw 'Concurrent VISSIM PID creation changed; no intervention or launch'
        }
    }
}
function Save-Status([string]$stage, [string]$arm, [object]$more) {
    [ordered]@{stage=$stage; arm=$arm; time=[DateTimeOffset]::Now.ToString('o'); pid=$PID; details=$more} |
      ConvertTo-Json -Depth 12 | Set-Content -LiteralPath (Join-Path $taskRoot 'status.json') -Encoding utf8
}
function Assert-CompletedReplay($plan, [string]$log) {
    $text = [IO.File]::ReadAllText($log)
    foreach ($marker in @('STAGE=SIM_DONE', ('SIM_SEC='+$protocol.end_sec), 'DECISIONS_FAILED=0',
        'OBSERVATION_FAILURES=0','SIGNAL_FAILURES=0','ACTION_FORMAT_FAILURES=0','COM_FAILURES=0')) {
        if ($text -notmatch ('(?m)^'+[regex]::Escape($marker)+'\r?$')) { throw "Incomplete replay cannot be reused: $log ($marker)" }
    }
    $watchlog=Join-Path $plan.out_dir 'WATCHDOG_PROGRESS.txt'
    if (-not (Select-String -LiteralPath $watchlog -Pattern (' OK '+[regex]::Escape($plan.name)+' attempt=1 ') -Quiet)) {
        throw 'Completed replay lacks successful watchdog exit'
    }
    $provenancePath=Join-Path $plan.out_dir ('run_provenance_'+$plan.name+'.json')
    $provenance=Get-Content -LiteralPath $provenancePath -Raw | ConvertFrom-Json
    if ($provenance.seed -ne $protocol.seed -or $provenance.sim_period_sec -ne $protocol.end_sec -or
        [IO.Path]::GetFullPath($provenance.workspace_root) -ne [IO.Path]::GetFullPath($FrozenTree)) { throw 'Completed replay context differs' }
    foreach ($pair in @(@('network','network'),@('tuning','tuning'),@('generated_vbs_config','vbs_config'))) {
        if ($provenance.files.($pair[0]).sha256 -cne (Hash $plan.($pair[1]))) { throw ('Completed replay source changed: '+$pair[0]) }
    }
    $native=Get-Content -LiteralPath (Join-Path $plan.out_dir 'native_network_performance.json') -Raw | ConvertFrom-Json
    if ($native.to_sec -ne $protocol.end_sec -or $native.read_at_sim_sec -ne $protocol.end_sec) { throw 'Completed replay endpoint missing' }
    [ordered]@{stage='completed_native_reused_pending_analysis';arm=$plan.arm;log_sha256=(Hash $log);
        provenance_sha256=(Hash $provenancePath);time=[DateTimeOffset]::Now.ToString('o');native_execution_validation_pending=$true} |
        ConvertTo-Json | Set-Content -LiteralPath (Join-Path $plan.out_dir 'resume_reuse.json') -Encoding utf8
}
function Wait-OwnedVissimExit($plan, [DateTime]$launchStarted) {
    $live=@(Get-UnreservedVissim @(Get-CimInstance Win32_Process | Where-Object { $_.Name -match '^Vissim' }))
    if (-not $live.Count) { return }
    $owned=Select-String -LiteralPath (Join-Path $plan.out_dir 'WATCHDOG_PROGRESS.txt') -Pattern ('VISSIM_PROCESS '+[regex]::Escape($plan.name)+' pid=(\d+)') | Select-Object -Last 1
    if (-not $owned) { throw 'VISSIM still open without owned-process receipt' }
    $ownedId=[int]$owned.Matches[0].Groups[1].Value
    if ($live.Count -ne 1 -or $live[0].ProcessId -ne $ownedId -or $live[0].CreationDate -lt $launchStarted) {
        throw 'Other VISSIM process is open; no intervention or new launch'
    }
    $process=Get-Process -Id $ownedId -ErrorAction SilentlyContinue
    if ($null -ne $process) {
        if ([Math]::Abs(($process.StartTime-$live[0].CreationDate).TotalSeconds) -gt .01) { throw 'VISSIM PID creation changed' }
        Write-Output ('WAIT_OWNED_VISSIM_EXIT pid='+$ownedId+' created='+$process.StartTime.ToString('o'))
        if (-not $process.WaitForExit(30000)) { throw 'Owned VISSIM teardown not complete; preserve results without killing or retrying' }
    }
}
