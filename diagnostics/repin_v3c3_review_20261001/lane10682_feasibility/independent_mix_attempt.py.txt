"""Independent seed43 native destination-population check; no fitting."""
import hashlib
import json
from pathlib import Path
from evaluation.controllers import obs150_contract as oc, offramp_routing as routing

HERE=Path(__file__).resolve().parent
ROOT=HERE.parents[2]
I=ROOT/'diagnostics/sdmpc_n31_20260924/integration_20260926'
pins={}
def read(path):
    raw=path.read_bytes();pins[str(path)]=hashlib.sha256(raw).hexdigest();return json.loads(raw)

def main():
    assert not (HERE/'independent_mix.json').exists()
    runtime=read(HERE/'native_cohorts.json')['route_runtime']
    folder=Path('D:/VISSIM_runs/20260929_seed43_observation2700/nc/decisions_sdmpc31_nc2700_s43')
    raws={t:read(folder/f'state_{t:06d}.json') for t in (2250,2400,2550,2700)}
    frames={t:read(Path(r['lane_plant_observation']['directory'])/f'frame_{t:06d}.json') for t,r in raws.items()}
    dets,h=oc.read_detector_csv(I/'selected/obs150/obs150_detectors_v2.csv');groups=oc.group_boundaries(dets)
    initial={v[0] for v in frames[2250]['vehicles'] if str(v[1]) in runtime['physical']}
    initial_known={v[0] for v in frames[2250]['vehicles'] if str(v[1]) in runtime['physical'] and v[6:8]==[1130.,1.]}
    source=[];off=[];tails=[]
    for t in (2400,2550,2700):
        bundle=oc.load_bundle(raws[t]);window=oc.assign_window(bundle.obs,bundle.mer_rows)
        for ref,dest in [('source:FW_E',source),('off_entry:10643',off)]:
            for det in groups[ref]:
                if window.tails[det.dcp_no]:tails.append(dict(end_sec=t,ref=ref,detector=det.dcp_no,count=window.tails[det.dcp_no]))
                dest.extend(dict(vehicle=e.veh,time_sec=e.t_entry) for e in window.entries[det.dcp_no])
    assert not any(x['ref']=='off_entry:10643' for x in tails)
    assert len({e['vehicle'] for e in off})==len(off)==76
    new_off={e['vehicle'] for e in off}-initial
    assert new_off<={e['vehicle'] for e in source}
    end_known=[];end_unknown=[]
    for v in frames[2700]['vehicles']:
        if v[0] in initial or str(v[1]) not in runtime['physical']:continue
        road,pos,cell=routing._position(runtime,v[1],v[3])
        if road!='FW_E' or pos>runtime['branches']['10643']['source_chain_m']:continue
        if v[6] is None:end_unknown.append(v[0])
        elif v[6:8]==[1130.,1.]:end_known.append(v[0])
    new_source={e['vehicle'] for e in source}-initial
    end_before={v[0] for v in frames[2700]['vehicles'] if v[0] not in initial and v[1]==74 and v[3]<40}
    prediction=read(I/'closedloop_recorded2250_lever450_trace10681_retained43/summary.json')['results']['held_actual']
    generated=prediction['control_area']['flow_counts']['origin:FW_E->freeway:FW_E']
    out=dict(status='COMPLETED_INDEPENDENT_NATIVE_MIX_CHECK',seed=43,start_sec=2250,end_sec=2700,
        native_off_entry=len(off),initial_known_count=len(initial_known),
        initial_known_exited=len(initial_known&{e['vehicle'] for e in off}),new_target_exited=len(new_off),
        new_target_retained=len(end_known),new_target_observed=len(new_off)+len(end_known),
        new_source_generation_lower_bound=len(new_source|end_before),source_tails=tails,
        unknown_new_mainline_before_exit=len(end_unknown),model_source_generated=generated,
        configured_target_probability=runtime['inputs']['FW_E']['weights']['10643'],
        expected_model_target_generated=generated*runtime['inputs']['FW_E']['weights']['10643'],
        source_pins=pins,new_native=0,new_fzp_scan=0,fit=0,
        limitation='This identifies realized destination population, not a replacement routing probability or proof of statistical independence.')
    (HERE/'independent_mix.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print({k:v for k,v in out.items() if k!='source_pins'})

def target10682():
    """Classify saved autonomous/native10682 cohorts; no forecast or fitting."""
    import gzip
    from collections import Counter
    outdir=HERE.parent/'entry10682'
    assert not (outdir/'assessment.json').exists()
    baseline=read(HERE.parent/'unrouted71/completion.json')
    for p,h in baseline['production7_unchanged'].items():
        assert hashlib.sha256(Path(p).read_bytes()).hexdigest()==h,p
    runtime=read(HERE/'native_cohorts.json')['route_runtime']
    off='10682';branch=runtime['branches'][off];cut=branch['source_chain_m']
    assert branch['freeway']=='FW_E'
    assert all(not m['weights'].get(off) for m in runtime['merges'].values())
    table=I/'selected/obs150/obs150_detectors_v2.csv'
    dets,digest=oc.read_detector_csv(table);pins[str(table)]=digest
    groups=oc.group_boundaries(dets)
    cases={}
    specs=[(43,2250,Path('D:/VISSIM_runs/20260929_seed43_observation2700/nc/decisions_sdmpc31_nc2700_s43'),
        'closedloop_recorded2250_lever450_trace10681_unrouted71_43'),
        (47,2700,Path('D:/VISSIM_runs/20260928_rm_observation2700_s47_v3/hold/decisions_sdmpc31_g_2700_hold_s47'),
        'closedloop_recorded2700_select_check_trace10681_unrouted71_47')]
    for seed,start,folder,pred_folder in specs:
        times=[start+i*150 for i in range(4)]
        raws={t:read(folder/f'state_{t:06d}.json') for t in times}
        frames={t:read(Path(r['lane_plant_observation']['directory'])/f'frame_{t:06d}.json') for t,r in raws.items()}
        initial={v[0]:v for v in frames[start]['vehicles'] if str(v[1]) in runtime['physical']}
        initial_all={v[0] for v in frames[start]['vehicles']}
        known={};uncertain={}
        for vid,v in initial.items():
            road,pos,cell=routing._position(runtime,v[1],v[3])
            if road!='FW_E' or pos>cut:continue
            weights=(routing._future_distribution(runtime,road,pos) if v[6] is None else
                routing._route_distribution(runtime,runtime['routes'][f'{int(v[6])}:{int(v[7])}'],road,pos))
            w=weights.get(off,0.)
            row=dict(vehicle=vid,cell=cell,lane=v[2],speed=v[4],position_m=pos,route=v[6:9],expected_weight=w)
            if w==1 and v[6] is not None:known[vid]=row
            elif w>0:uncertain[vid]=row
        events=[];source=[];tails=[];removed={}
        for t in times[1:]:
            bundle=oc.load_bundle(raws[t]);window=oc.assign_window(bundle.obs,bundle.mer_rows)
            for name,output in [(f'off_entry:{off}',events),('source:FW_E',source)]:
                for det in groups[name]:
                    if window.tails[det.dcp_no]:tails.append(dict(end_sec=t,ref=name,detector=det.dcp_no,count=window.tails[det.dcp_no]))
                    output.extend(dict(vehicle=e.veh,time_sec=e.t_entry,lane=det.lane) for e in window.entries[det.dcp_no])
            for e in oc.window_removals(bundle.err_rows,start,start+450):removed[e['raw_line_sha256']]=e
        assert not any(e['ref']==f'off_entry:{off}' for e in tails)
        assert len({e['vehicle'] for e in events})==len(events)
        source_ids={e['vehicle'] for e in source}
        new_off={e['vehicle'] for e in events}-set(initial)
        # If a different origin exists, retain it as unresolved instead of
        # silently calling every noninitial mainline vehicle a new source.
        no_source=sorted(new_off-source_ids)
        snaps=[]
        for t in times:
            lookup={v[0]:v for v in frames[t]['vehicles']}
            exited={e['vehicle'] for e in events if e['time_sec']<=t+.1}
            known_remaining=[];known_other=[];new_target=[]
            for vid,first in known.items():
                if vid in exited:continue
                v=lookup.get(vid)
                if v is not None and str(v[1]) in runtime['physical']:
                    road,pos,cell=routing._position(runtime,v[1],v[3])
                    known_remaining.append(dict(vehicle=vid,road=road,cell=cell,lane=v[2],speed=v[4],position_m=pos,route=v[6:9]))
                else:known_other.append(dict(vehicle=vid,current=v))
            for vid,v in lookup.items():
                if vid in initial or str(v[1]) not in runtime['physical'] or v[6] is None:continue
                road,pos,cell=routing._position(runtime,v[1],v[3])
                if road!='FW_E' or pos>cut:continue
                route=runtime['routes'][f'{int(v[6])}:{int(v[7])}']
                if route['target']==off:new_target.append(dict(vehicle=vid,cell=cell,lane=v[2],speed=v[4],position_m=pos,source_seen=vid in source_ids))
            snaps.append(dict(time_sec=t,initial_known_exit=len(set(known)&exited),
                initial_uncertain_exit=len(set(uncertain)&exited),initial_other_exit=len((set(initial)-set(known)-set(uncertain))&exited),
                initial_known_remaining=known_remaining,initial_known_unaccounted=known_other,
                new_target_exit=len(exited-set(initial)),new_target_remaining=new_target))
        summary=read(I/pred_folder/'summary.json');prediction=summary['results']['held_actual']
        tracepath=I/pred_folder/'held_actual_RM_C10681_trace.json.gz'
        raw=tracepath.read_bytes();pins[str(tracepath)]=hashlib.sha256(raw).hexdigest()
        trace=json.loads(gzip.decompress(raw))['offramp_network_diagnostics']
        assert len(trace['route_states'])==4 and len(trace['states'])==4
        model=[];first=None
        for stock,port in zip(trace['route_states'],trace['states']):
            c=Counter()
            for road,cells in stock['inventory']['cells'].items():
                for cell,row in enumerate(cells):
                    for label,n in row.items():
                        if label.endswith('|'+off):c[label]+=n
            if first is None:first=c.copy()
            initial_known_drop=first['observed_route:1130:3|route:1130:3|10682']-c['observed_route:1130:3|route:1130:3|10682']
            uncertain_drop=sum(n-c[label] for label,n in first.items() if label!='observed_route:1130:3|route:1130:3|10682')
            admitted=port['ports'][off]['admitted']
            model.append(dict(time_sec=stock['time_sec'],initial_known_exit=initial_known_drop,
                initial_uncertain_exit=uncertain_drop,future_exit=admitted-initial_known_drop-uncertain_drop,
                future_remaining=c['expected_input:1098|route:1130:3|10682'],stocks=dict(c),total_entry=admitted))
        assert abs(first['observed_route:1130:3|route:1130:3|10682']-len(known))<1e-9
        assert abs(sum(first.values())-len(known)-sum(v['expected_weight'] for v in uncertain.values()))<1e-9
        generated=prediction['control_area']['flow_counts']['origin:FW_E->freeway:FW_E']
        probability=runtime['inputs']['FW_E']['weights'][off]
        assert abs(model[-1]['future_exit']+model[-1]['future_remaining']-generated*probability)<1e-7
        cases[str(seed)]=dict(start_sec=start,end_sec=start+450,native_total_entry=len(events),
            native_initial_known=list(known.values()),native_initial_uncertain=list(uncertain.values()),
            native_snapshots=snaps,native_entries=events,source_tails=tails,new_exit_without_source_witness=no_source,
            initial_known_removals=[e for e in removed.values() if e['vehicle_id'] in known],
            model_snapshots=model,model_source_generated=generated,model_target_probability=probability,
            model_expected_new_target=generated*probability,model_entry=trace['states'][-1]['ports'][off]['admitted'],
            trace_source=pred_folder,trace_candidate='unrouted71 trial, restored not adopted; reuse its recorded route states. Local71-only change alters10682 admission by <0.05veh.',
            native_not_initial_network_entry=sum(vid not in initial_all for vid in new_off))
        print(seed,'native total',len(events),'known',len(known),'known exit',snaps[-1]['initial_known_exit'],
            'new exit/remain',snaps[-1]['new_target_exit'],len(snaps[-1]['new_target_remaining']),
            'model knownexit/newexit/remain',model[-1]['initial_known_exit'],model[-1]['future_exit'],model[-1]['future_remaining'],
            'targetgen',generated*probability,'missing sources',no_source,flush=True)
    for p,h in baseline['production7_unchanged'].items():assert hashlib.sha256(Path(p).read_bytes()).hexdigest()==h,p
    out=dict(status='completed_saved_native_model10682_cohort_decomposition',cases=cases,source_pins=pins,
        new_native=0,new_forecasts=0,new_fzp_scans=0,new_fits=0,
        limitations=['Future observations classify realized outcomes only; not autonomous inputs.',
            'Future destination count is a confirmed lower bound when some routes are unassigned.',
            'Population and residual-count decomposition is accounting, not independent causal attribution.',
            'Native epochs contain0.1s offset; tail and missing events retained.'])
    (outdir/'assessment.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n',encoding='utf8')


if __name__=='__main__':
    import sys
    if sys.argv[1:]==['--target10682']:target10682()
    elif not sys.argv[1:]:main()
    else:raise SystemExit('Expected no arguments or --target10682')
