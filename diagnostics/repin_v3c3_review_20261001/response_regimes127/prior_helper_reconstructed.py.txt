def read(p):
    data=Path(p).read_bytes()
    return json.loads(gzip.decompress(data) if str(p).endswith('.gz') else data)


def save(p,x):
    Path(p).write_text(json.dumps(x,ensure_ascii=False,indent=2,allow_nan=False)+'\n',encoding='utf-8')


def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def losses(rows,z):
    """Calibration score; the controller's Omega objective is not modified."""
    absolute=[]; local=[]; response=[]; pairs=[]
    for row in rows:
        a,p=row['actual'],row['predicted']
        terms=[((p[k]-a[k])/s)**2 for k,s in [('ttt',5.),('ramp_ttt',1.),('off_ttt',1.),('end_n',40.),('exits',30.)]]
        terms += [((p['merges'][r]-a['merges'][r])/20.)**2 for r in a['merges']]
        absolute.append(sum(terms)/len(terms))
        local.append(sum((row['local_rmse'][k]/s)**2 for k,s in [('n',8.),('v',20.),('q',800.)])/3)
    for case in sorted({x['case'] for x in rows}):
        by={x['arm']:x for x in rows if x['case']==case}
        # All six contrasts preserve both incremental VSL effects and RM effects.
        for a,b in itertools.combinations(sorted(by),2):
            left,right=by[a],by[b]
            actual={k:right['actual'][k]-left['actual'][k] for k in ('ttt','ramp_ttt','end_n','exits')}
            pred={k:right['predicted'][k]-left['predicted'][k] for k in actual}
            terms=[((pred[k]-actual[k])/max(s,abs(actual[k])))**2 for k,s in [('ttt',.5),('ramp_ttt',.5),('end_n',15.),('exits',15.)]]
            terms += [((right['predicted']['merges'][r]-left['predicted']['merges'][r]-(right['actual']['merges'][r]-left['actual']['merges'][r]))/max(10.,abs(right['actual']['merges'][r]-left['actual']['merges'][r])))**2 for r in left['actual']['merges']]
            response.append(sum(terms)/len(terms))
            pairs.append(dict(case=case,left=a,right=b,actual=actual,predicted=pred,
                              meaningful=abs(actual['ttt'])>=.5,sign_correct=actual['ttt']*pred['ttt']>0))
    avg=lambda xs:sum(xs)/len(xs) if xs else 0.
    ans=dict(absolute=avg(absolute),local=avg(local),response=avg(response),pairs=pairs,
             prior=avg([(v-1.)**2 for v in z]))
    ans['objective']=ans['absolute']+.25*ans['local']+2*ans['response']+.1*ans['prior']
    return ans

        destination=I/('closedloop_recorded2700_lever450_'+label);assert not destination.exists()
        jobs.append(dict(name=name,argv=argv,output=str(destination)))
    pins={str(Path(commands[0])):sha(commands[0]),str(original_tuning):sha(original_tuning),str(manifest):sha(manifest),str(candidate):sha(candidate)}
    save(out/'protocol.json',dict(reason='43 late boundary realization differs;73 independently observed equal918 source counts. Equal totals alone are not proof of identical arrival times. Check same fixed candidate without retuning.',
        budget=dict(baseline450=2,candidate450=2,fits=0,new_native=0,FZP=0),candidate=[1.,.65,1.],
        scope='WholeOmegaFW+protectedurban450s from native73 common2700 state; same past head/VSL history and executed commands. Externalwaitingdiagnostic.',
        source=read(R/'replication_s73_95/response_assessment95.json')['actual_delta_omega_veh_h'],
        jobs=jobs,pins=pins,protected_sha256=prior['protected_sha256'],STOP=prior['STOP'],
        gate='Fresh baseline must reproduce95 physical results exactly; no future traffic states; same frozen126 coefficients; preserve actual gain/loss and do not tune on73.'))
    started=time.perf_counter()
    try:
        for job in jobs:
            save(out/'status.json',dict(status='running',job=job['name']))
            with (out/(job['name']+'.log')).open('w',encoding='utf-8') as f:
                child=subprocess.run([sys.executable]+job['argv'],cwd=ROOT,stdout=f,stderr=subprocess.STDOUT,
                    creationflags=subprocess.BELOW_NORMAL_PRIORITY_CLASS,check=False)
            assert child.returncode==0,(job['name'],child.returncode)
            summary=read(Path(job['output'])/'summary.json')
            assert summary['future_observation_inputs'] is False and summary['native_started'] is False and summary['optimizer_iterations']==0
            if job['name']=='baseline':
                old=read(I/'closedloop_recorded2700_lever450_s73_replication95/summary.json')
                fields=('ttt_omega_veh_h','tracked_outside_residence_veh_h','cost_by_stock','outside_cost_by_stock','ramps','commands','physical_cell_states','control_area')
                checks={arm:{k:values[k]==old['results'][arm][k] for k in fields} for arm,values in summary['results'].items()}
                save(out/'baseline_parity.json',checks);assert all(all(row.values()) for row in checks.values())
        summaries={j['name']:read(Path(j['output'])/'summary.json') for j in jobs}
        deltas={name:s['results']['release_vsl90_actual']['ttt_omega_veh_h']-s['results']['release_actual']['ttt_omega_veh_h'] for name,s in summaries.items()}
        actual=read(R/'replication_s73_95/response_assessment95.json')['actual_delta_omega_veh_h']
        save(out/'assessment.json',dict(actual_delta_omega_veh_h=actual,predicted_deltas=deltas,
            candidate_error=abs(deltas['candidate']-actual),correct_sign=deltas['candidate']*actual>0,
            magnitude_gate=abs(deltas['candidate']-actual)<=max(.5,.5*abs(actual)),
            new_native=0,new_fits=0,production_adopted=False,goal_complete=False))
        save(out/'status.json',dict(status='complete_not_adopted',forecasts=4,elapsed_sec=time.perf_counter()-started))
        print(json.dumps(dict(actual=actual,predicted=deltas)),flush=True)
    except BaseException as exc:
        save(out/'status.json',dict(status='failed',error=repr(exc),elapsed_sec=time.perf_counter()-started));raise
    finally:
        for p,h in {**prior['protected_sha256'],**pins}.items():assert sha(p)==h,p
        assert sha(prior['STOP']['path'])==prior['STOP']['sha256']
        save(out/'preservation.json',dict(core=True,inputs=True,STOP=True))
