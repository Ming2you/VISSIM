"""Causal wiring probe using the canonical adapter and endpoint; no VISSIM/fitting.

Intercept after runtime/SDMPC configuration and forecast construction, before
optimization. Perturb only the redundant exogenous ramp-arrival field. This
does not perturb physical urban route arrivals or prove their accuracy.
"""
from __future__ import annotations

import copy
import csv
import gzip
import hashlib
import json
import os
from contextlib import contextmanager, nullcontext
import sys
import time
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[3]
HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT))


def probe_controller_derivatives(captured, reference, coord, evaluate, derivative, quantities,
                                 output, sources, resume=None, protocol_path=None):
    """One actual-controller AD and bounded independent continuous scalar checks.

    Infinitesimal FD commands bypass native quantization, exactly like the AD
    seeding. They diagnose two columns, not feasible commands or a new optimum.
    """
    import pickle
    import numpy as np
    from evaluation.controllers import sdmpc_sequence as sequence
    from evaluation.controllers.sdmpc_tangent import checked_matrices
    from evaluation.controllers.sdmpc_tangent_surrogate import Query, token

    protocol_path = protocol_path or HERE/'closedloop9000_d4e2_analysis/controller_derivative_check/protocol.json'
    protocol = json.loads(protocol_path.read_bytes())
    query = derivative.__self__
    assert isinstance(query, Query) and query.stats()['total_rollouts'] == 0
    cfg = captured['cfg']
    start_sec = protocol.get('start_sec', 4500)
    assert captured['state'].time_sec == start_sec and cfg.mpc.horizon_steps == 3
    anchor_meters = protocol.get('anchor_meter_green', {'RM_C10484': (8., 6., 4.)})
    continuous_offset = protocol.get('continuous_anchor_offset')
    branch_probe=protocol.get('branch_probe')
    full_witness=protocol.get('full_primal_witness',False)
    assert type(full_witness) is bool
    if full_witness:
        assert (resume is None and branch_probe is None and continuous_offset is None
                and protocol['axes']==[dict(kind='meter',key='RM_C10490',block=0,steps=[])])
    if branch_probe is not None:
        assert resume is None and continuous_offset is None and branch_probe in (
            dict(kind='meter',key='RM_C10490',block=0,max_records=200000),
            dict(kind='offset',key='SC1001',block=0,max_records=200000))
        assert protocol['axes']==[dict(kind=branch_probe['kind'],key=branch_probe['key'],block=0,steps=[])]
    if continuous_offset is not None:
        assert (start_sec == 2700 and resume is None and not anchor_meters
                and continuous_offset in (
                    dict(key='RM_C10490', block=0, delta_sec=-.001),
                    dict(key='RM_C10490', block=0, delta_sec=.001))
                and protocol['axes'] == [dict(kind='meter', key='RM_C10490', block=0, steps=[.0001])])
    if start_sec != 4500:
        assert (protocol['schema'] == 'recorded-controller-derivatives/v1'
                and start_sec == 2700 and protocol['seed'] == 47
                and protocol['horizon_sec'] == 450 and not anchor_meters
                and protocol['max_ad_rollouts'] == 1
                and protocol['max_independent_scalar_rollouts'] == (0 if branch_probe else 1 if full_witness
                    else 1+2*sum(len(spec['steps']) for spec in protocol['axes']))
                and protocol['max_independent_scalar_rollouts'] <= 13
                and protocol['no_optimization'] and protocol['no_native']
                and not protocol['future_observation_inputs'] and protocol['fit'] == 0)
    z = coord.encode(reference)
    for j,axis in enumerate(coord.axes):
        if axis['kind']=='meter' and axis['key'] in anchor_meters:
            z[j] = (anchor_meters[axis['key']][axis['block']]-axis['reference_value'])/axis['scale']
    anchor = coord.decode(z, reference)
    coord.validate(anchor)
    for k,action in enumerate(sequence.actions(anchor,3)):
        assert action.green_times == reference.green_times and action.offsets == reference.offsets
        assert action.vsl == reference.vsl
        for ramp in cfg.network.ramps:
            key = 'rw_meter_green_'+ramp
            expected = anchor_meters[ramp][k] if ramp in anchor_meters else reference.diagnostics[key]
            assert action.diagnostics[key] == expected
    if continuous_offset is not None:
        # Validate the actual command above, then perturb only this diagnostic
        # linearization point. It must never pass through native quantization.
        controls = sequence.actions(anchor, 3)
        controls[0].diagnostics['rw_meter_green_RM_C10490'] += continuous_offset['delta_sec']
        shifted = sequence.pack(controls)
        delta = coord.encode(shifted)-coord.encode(anchor)
        j = next(i for i,a in enumerate(coord.axes)
                 if (a['kind'],a['key'],a['block']) == ('meter','RM_C10490',0))
        expected = np.zeros(len(coord.axes))
        expected[j] = continuous_offset['delta_sec']/coord.axes[j]['scale']
        assert np.max(abs(delta-expected)) < 1e-12
        anchor = shifted
    source_pins = {str(p):hashlib.sha256(p.read_bytes()).hexdigest()
                   for p in (*sources,protocol_path,Path(__file__))}
    history_receipt = output/'vsl_history_replay.json'
    source_pins.update(json.loads(history_receipt.read_bytes())['original_files'])
    selected_axes = []
    trials = [anchor]
    trial_names = ['independent_anchor']
    # Physical units: seconds of meter GREEN and km/h of the VSL command.
    # Jacobians are w.r.t. normalized z, so h_z = h_physical / axis.scale.
    for spec in protocol['axes']:
        matches = [(j,a) for j,a in enumerate(coord.axes)
                   if all(a[key]==spec[key] for key in ('kind','key','block'))
                   and ('owner' not in spec or a['owner']==spec['owner'])]
        assert len(matches)==1
        j,axis = matches[0]
        row = dict(axis, index=j, steps=spec['steps'],
                   step_unit=('green_sec' if axis['kind'] in ('meter','green') else
                              'offset_sec' if axis['kind']=='offset' else 'km_h'), trials=[])
        for step in spec['steps']:
            indices=[]
            for sign in (-1,1):
                controls=sequence.actions(anchor,3)
                control=controls[axis['block']]
                if axis['kind']=='green':
                    for phase,factor in coord.green_vectors[j].items():
                        control.green_times[axis['owner']+'_'+phase] += sign*step*factor
                elif axis['kind']=='offset':
                    control.offsets[axis['key']] += sign*step
                elif axis['kind']=='meter':
                    control.diagnostics['rw_meter_green_'+axis['key']] += sign*step
                elif axis['kind']=='vsl':
                    speed=control.vsl[axis['key']]+sign*step
                    for cell,head in enumerate(cfg.network.freeway_vsl_zone_head_of_cell[axis['owner']]):
                        if head==axis['head']:
                            control.vsl[f"{axis['owner']}__seg{cell}"]=speed
                    control.vsl[axis['owner']]=min(control.vsl[f"{axis['owner']}__seg{i}"]
                        for i in range(len(cfg.network.freeway_vsl_zone_head_of_cell[axis['owner']])))
                else:
                    raise ValueError('Unknown diagnostic control axis')
                action=sequence.pack(controls)
                delta=coord.encode(action)-coord.encode(anchor)
                expected=np.zeros(len(coord.axes));expected[j]=sign*step/axis['scale']
                assert np.max(abs(delta-expected))<1e-12
                indices.append(len(trials));trials.append(action)
                trial_names.append(f"{axis['key']}_block{axis['block']}_{sign*step:+g}")
            row['trials'].append(dict(step_physical=step,step_z=step/axis['scale'],indices=indices))
        selected_axes.append(row)
    assert (branch_probe and len(trials)==1) or len(trials)==protocol['max_independent_scalar_rollouts']<=13
    def write(name,value):
        (output/name).write_text(json.dumps(value,ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')
    write('protocol.json',dict(protocol,files=source_pins,axes=selected_axes,
        command_scope=('Offset anchor and FD points are continuous diagnostics, never applied'
                       if continuous_offset else 'Native-valid anchor only; FD points are continuous diagnostics, never applied'),
        scalar_workers=3,optimizer_iterations=0,native_started=False))
    started=time.perf_counter()
    owners=(*coord.owners,'PASSIVE_OMEGA')
    def costs(item):return np.array([item['sdmpc_omega_partition']['costs'][p] for p in owners])
    if full_witness:
        # Use the existing worker's independent scalar/full-state witness path.
        # This is outside Query reuse: an AD primal is never its own witness.
        from evaluation.controllers.sdmpc_tangent import _evaluate_single
        request=pickle.loads(query.template)
        assert 'surrogate_evaluation' not in request
        request['action']=anchor
        # Retain full audit records for the witness. The streamed-record and
        # central-constraint observers only support single surrogate passes;
        # neither participates in traffic evolution or cost/resource values.
        diagnostic_options=request['owned'][0].cfg.network.sdmpc_options
        observer_flags={name:diagnostic_options[name]
                        for name in ('prediction_hotpath','central_multiplier')}
        assert all(value is True for value in observer_flags.values())
        diagnostic_options.update({name:False for name in observer_flags})
        query.check_budget('diagnostic_full_primal_witness')
        result=_evaluate_single(request)
        assert (result['complete_primal_state_match']
                and result['max_primal_state_error']<=protocol['primal_abs_tolerance']
                and result['scalar_rollouts']==result['tangent_rollouts']==1)
        prior=json.loads(Path(protocol['reference_ad_columns']).read_bytes())
        errors=[float(np.max(abs(np.array(result[name])-prior[other])))
                for name,other in (('costs','costs'),('resources','resource_values'))]
        for old in prior['columns']:
            j=old['axis']['index']
            errors.extend(float(np.max(abs(np.array(result[name])[:,j]-old[other])))
                          for name,other in (('cost_jacobian','costs'),('resource_jacobian','resources')))
        assert max(errors)<1e-7
        assert all(hashlib.sha256(Path(p).read_bytes()).hexdigest()==sha for p,sha in source_pins.items())
        write('witness.json',result)
        write('summary.json',dict(status='complete_independent_primal_witness',
            wall_sec=time.perf_counter()-started,prior_cost_resource_gradient_max_error=max(errors),
            max_primal_state_error=result['max_primal_state_error'],full_primal_trajectory_audited=True,
            diagnostic_observer_flags_disabled=observer_flags,
            new_ad_rollouts=1,new_scalar_rollouts=1,optimizer_iterations=0,native_started=False,
            files=source_pins,calibration_changed=False,
            scope='Full scalar/AD primal trajectory; current RM10490 Jacobian matches prior receipt. Not an all-axis finite-difference or native-gain certificate.'))
        print(json.dumps(dict(status='complete_independent_primal_witness',max_primal_state_error=result['max_primal_state_error'])),flush=True)
        return
    if resume is None:
        runner=query.runner
        if branch_probe is not None:
            query.runner=lambda request:runner(dict(request,diagnostic_branch_probe=branch_probe))
        try:item=evaluate([anchor],derivatives=True)[0]
        finally:query.runner=runner
        receipt=derivative(anchor)
        gradient,resource_gradient,fallback=checked_matrices(receipt,costs(item),quantities(anchor,item),
            coord.axes,protocol['primal_abs_tolerance'],allow_surrogate=True)
        assert not fallback and query.stats()['tangent_rollouts']==protocol['max_ad_rollouts']==1
        ad_data=dict(owners=owners,resources=['NP_veh','NUF_veh_h'],
            columns=[dict(axis=a,costs=gradient[:,a['index']].tolist(),
                          resources=resource_gradient[:,a['index']].tolist()) for a in selected_axes],
            costs=costs(item).tolist(),resource_values=quantities(anchor,item).tolist(),
            trace=receipt['trace'],derivative_scope=receipt['derivative_scope'],
            transformed_source_sha256=receipt['transformed_source_sha256'])
        ad_stats=query.stats()
    else:
        old=json.loads((resume/'protocol.json').read_bytes())
        ad_data=json.loads((resume/'ad_columns.json').read_bytes())
        failed=json.loads((resume/'unused_action.joint.json').read_bytes())
        ad_stats=failed['physical_response_cache']
        assert not failed['completed'] and ad_stats['tangent_rollouts']==1 and ad_stats['scalar_rollouts']==0
        assert ad_stats['rows'][0]['action_token']==token(anchor)
        assert old['axes']==selected_axes and ad_data['owners']==list(owners)
        # The only changed source is this harness's failed witness setup; all
        # input/model/AD sources must retain their original bytes.
        for path,sha in {**old['files'],**ad_data['transformed_source_sha256']}.items():
            p=Path(path)
            if p.resolve()==Path(__file__).resolve():
                p=HERE/'executed_sources'/('probe_selected_arrival_path_'+sha+'.py')
            assert hashlib.sha256(p.read_bytes()).hexdigest()==sha,path
        gradient=np.full((len(owners),len(coord.axes)),np.nan)
        resource_gradient=np.full((2,len(coord.axes)),np.nan)
        for column in ad_data['columns']:
            j=column['axis']['index']
            gradient[:,j]=column['costs'];resource_gradient[:,j]=column['resources']
    write('ad_columns.json',ad_data)
    if branch_probe is not None:
        prior_path=Path(protocol['reference_ad_columns']);prior=json.loads(prior_path.read_bytes())
        prior_column=next(c for c in prior['columns'] if all(
            c['axis'][k]==branch_probe[k] for k in ('kind','key','block')))
        primal_error=max(float(np.max(abs(np.array(ad_data['costs'])-prior['costs']))),
                         float(np.max(abs(np.array(ad_data['resource_values'])-prior['resource_values']))))
        gradient_error=max(float(np.max(abs(np.array(ad_data['columns'][0]['costs'])-prior_column['costs']))),
                           float(np.max(abs(np.array(ad_data['columns'][0]['resources'])-prior_column['resources']))))
        expected_change=protocol.get('expected_gradient_change')
        assert expected_change in (None, 'zero_active_ramp_transfer', 'shared_approach_cohort_cleanup')
        assert primal_error<1e-7
        if expected_change is None:
            assert gradient_error<1e-7
        error=max(primal_error,gradient_error)
        write('branch_report.json',receipt['diagnostic_branch_report'])
        assert all(hashlib.sha256(Path(p).read_bytes()).hexdigest()==sha for p,sha in source_pins.items())
        write('summary.json',dict(status='complete_branch_diagnostic',wall_sec=time.perf_counter()-started,
            prior_cost_resource_gradient_max_error=error,reference_ad_sha256=hashlib.sha256(prior_path.read_bytes()).hexdigest(),
            prior_primal_max_error=primal_error,prior_gradient_max_change=gradient_error,
            expected_gradient_change=expected_change,
            ad_query=ad_stats,new_ad_rollouts=1,new_scalar_rollouts=0,optimizer_iterations=0,native_started=False,
            files=source_pins,calibration_changed=False,full_primal_trajectory_audited=False))
        print(json.dumps(dict(status='complete_branch_diagnostic',prior_parity_error=error,wall_sec=time.perf_counter()-started)),flush=True)
        return
    # A fresh query cannot reuse the AD primal as an alleged scalar witness.
    independent=Query(pickle.loads(query.template),query.check_budget,query.emit,runner=query.runner)
    # pickle's alias memo may change on load/repack despite equal operands.
    # Retain the exact original request bytes and their original context token.
    independent.template=query.template
    independent.context_token=query.context_token
    scalar=[]
    try:
        for begin in range(0,len(trials),3):
            scalar.extend(independent.evaluate(trials[begin:begin+3],derivatives=False))
            write('scalar_responses.json',dict(names=trial_names[:len(scalar)],results=scalar))
    finally:
        independent.close()
    f0=costs(scalar[0]);q0=quantities(anchor,scalar[0])
    primal_cost_error=float(np.max(abs(np.asarray(ad_data['costs'])-f0)))
    primal_resource_error=float(np.max(abs(np.asarray(ad_data['resource_values'])-q0)))
    rows=[]
    for axis in selected_axes:
        j=axis['index']
        comparisons=[]
        for trial in axis['trials']:
            minus,plus=trial['indices'];h=trial['step_z']
            fm,fp=costs(scalar[minus]),costs(scalar[plus])
            qm,qp=(quantities(trials[idx],scalar[idx]) for idx in (minus,plus))
            dc,dq=(fp-fm)/(2*h),(qp-qm)/(2*h)
            close_c=np.isclose(dc,gradient[:,j],atol=protocol['cost_gradient_atol'],rtol=protocol['gradient_rtol'])
            close_q=np.isclose(dq,resource_gradient[:,j],atol=protocol['resource_gradient_atol'],rtol=protocol['gradient_rtol'])
            comparisons.append(dict(step_physical=trial['step_physical'],step_z=h,
                central_costs=dc.tolist(),left_costs=((f0-fm)/h).tolist(),right_costs=((fp-f0)/h).tolist(),
                central_resources=dq.tolist(),left_resources=((q0-qm)/h).tolist(),right_resources=((qp-q0)/h).tolist(),
                omega_ad_per_z=float(sum(gradient[:,j])),omega_fd_per_z=float(sum(dc)),
                max_cost_error=float(np.max(abs(dc-gradient[:,j]))),
                max_resource_error=float(np.max(abs(dq-resource_gradient[:,j]))),
                cost_matches=close_c.tolist(),resource_matches=close_q.tolist(),
                all_match=bool(np.all(close_c) and np.all(close_q))))
        rows.append(dict(axis=axis,comparisons=comparisons))
    pins_unchanged=all(hashlib.sha256(Path(p).read_bytes()).hexdigest()==sha for p,sha in source_pins.items())
    assert pins_unchanged
    stats=independent.stats()
    assert stats['scalar_rollouts']==len(trials) and stats['tangent_rollouts']==0
    report=dict(status='complete',start_sec=start_sec,horizon_sec=450,files=source_pins,
        primal_cost_max_error=primal_cost_error,primal_resource_max_error=primal_resource_error,
        primal_outputs_match=max(primal_cost_error,primal_resource_error)<=protocol['primal_abs_tolerance'],
        full_primal_trajectory_audited=False,rows=rows,
        all_checked_gradients_match=all(c['all_match'] for row in rows for c in row['comparisons']),
        ad_query=ad_stats,independent_scalar_query=stats,wall_sec=time.perf_counter()-started,
        reused_ad_from=str(resume) if resume else None,new_ad_rollouts=query.stats()['tangent_rollouts'],
        optimizer_iterations=0,native_started=False,calibration_changed=False,files_unchanged=pins_unchanged,
        scope='Selected continuous columns at one anchor; not all-axis AD certification, an optimizer result, or native gain validation')
    write('summary.json',report)
    print(json.dumps({k:report[k] for k in ('status','primal_outputs_match','all_checked_gradients_match','wall_sec')}),flush=True)


def diagnostic_source_forecast(forecast, case, document):
    """Future-input ablation only. Never used by an optimizer or native runner."""
    import math
    result = copy.deepcopy(forecast[:3])
    if document is None:
        return result
    pair = document.get('schema') == 'diagnostic-native-mainline-pair/v1'
    if (document.get('schema') not in ('diagnostic-native-mainline/v1', 'diagnostic-native-mainline-pair/v1')
            or document.get('road') != 'FW_E' or document.get('start_sec') != (2700. if pair else 2250.)
            or document.get('interval_sec') != 150. or document.get('blocks') != 3
            or document.get('future_observation_inputs') is not True
            or document.get('optimizer_allowed') is not False
            or document.get('mode') not in ('common_nc', 'per_arm')):
        raise ValueError('Invalid future-source diagnostic contract')
    if pair and (document.get('autonomous_prediction') is not False
            or document.get('calibration_allowed') is not False
            or document.get('forecast_budget') != 2 or document.get('new_native_runs') != 0
            or document.get('fit') != 0 or document.get('mode') != 'per_arm'):
        raise ValueError('The saved pair is conditional diagnosis only, with two fixed forecasts')
    profiles = document['profiles_veh_h']
    expected = {'held_actual', 'selected'} if pair else {'held_actual', 'rm', 'vsl', 'both'}
    if set(profiles) != expected or len(result) != 3:
        raise ValueError('Future-source diagnostic needs exactly its fixed command cases')
    if document['mode'] == 'common_nc' and any(v != profiles['held_actual'] for v in profiles.values()):
        raise ValueError('Common-source diagnostic must hold the same input in every case')
    rates = profiles[case]
    if len(rates) != 3 or any(not math.isfinite(float(q)) or q < 0 for q in rates):
        raise ValueError('Invalid diagnostic source rates')
    for step, rate in zip(result, rates):
        step.freeway_mainline['FW_E'] = float(rate)
    return result


@contextmanager
def observe_first_interval_speed_terms(start, rows, *, horizon_sec=150, cells=(0,4,*range(9,26))):
    """Observe the installed scalar equations; restore all hooks on exit."""
    if type(horizon_sec) is not int or not 0 < horizon_sec <= 750:
        raise ValueError('Diagnostic observation horizon must be1..750seconds')
    from evaluation.controllers import area_freeway_accounting as accounting
    from evaluation.controllers import vissim_stackelberg_adapter as adapter
    from evaluation.controllers.freeway_fd import state_response_coefficients
    mn = accounting._mn
    step, segment, update = accounting._freeway_substep_events, mn.segment_vsl, mn.metanet_speed_update_kmh
    active = {}

    def observe_segment(control, road, cell, cfg, **kwargs):
        result = segment(control,road,cell,cfg,**kwargs)
        active['address'] = (road,cell)
        return result

    def observe_update(speed, upstream, rho, down, desired, dt, length, tau, nu, kappa, minimum):
        ctx = copy.deepcopy(adapter._FW_SEG_CTX)
        address = active.get('address')
        result = update(speed,upstream,rho,down,desired,dt,length,tau,nu,kappa,minimum)
        if (active.get('time',start+horizon_sec) >= start+horizon_sec or address is None or address[0]!='FW_E'
                or address[1] not in cells):
            return result
        assert ctx['armed']
        p = ctx['p']
        length = p.get('segment_length_km',length)
        tau = p.get('metanet_tau_h',tau)
        kappa = p.get('metanet_kappa_veh_km_lane',kappa)
        tau,nu = state_response_coefficients(ctx.get('state_response',{}),speed,desired,rho,down,
                                             ctx['response_rho_crit'],tau,nu)
        relaxation = dt/max(tau,1e-9)*(desired-speed)
        convection = dt/max(length,1e-9)*speed*(upstream-speed)
        anticipation = -nu*dt/(max(tau,1e-9)*max(length,1e-9))*(down-rho)/max(rho+kappa,1e-9)
        before_drop = max(minimum,speed+relaxation+convection+anticipation)
        drop = (ctx['phi']*dt*ctx['dlam']*max(rho,0.)*speed**2 /
                (max(length,1e-9)*max(ctx['lanes'],1e-9)*max(p.get('rho_crit',ctx['rho_crit_link']),1e-9)))
        assert abs(max(minimum,before_drop-drop)-result) < 1e-8
        rows.append(dict(time_sec=active['time'],cell=address[1],speed_before=speed,rho=rho,
            downstream_rho=down,desired=desired,relaxation=relaxation,convection=convection,
            anticipation=anticipation,lane_drop_raw=drop,after_speed_equation=result,
            phi=ctx['phi'],delta_lanes=ctx['dlam'],length_km=length,tau_sec=tau*3600,nu=nu))
        return result

    def observe_step(state, control, demand, cfg, *args, **kwargs):
        active['time'] = state.time_sec
        mark = len(rows)
        result = step(state,control,demand,cfg,*args,**kwargs)
        for row in rows[mark:]:
            row['speed_final'] = state.freeway_speed['FW_E'][row['cell']]
            row['post_equation_change'] = row['speed_final']-row['after_speed_equation']
        return result

    mn.segment_vsl, mn.metanet_speed_update_kmh = observe_segment,observe_update
    accounting._freeway_substep_events = observe_step
    try:
        yield
    finally:
        mn.segment_vsl, mn.metanet_speed_update_kmh = segment,update
        accounting._freeway_substep_events = step


def first_interval_response(state, point, cfg):
    """Export an existing rollout's executed 150s prefix; no future inputs."""
    end = point.states[0]
    t0, t1 = state.time_sec, end.time_sec
    assert t1-t0 == 150
    response = point.control_area_response

    def prefix(rows):
        assert not any(r['start_sec'] < t1 < r['end_sec'] for r in rows)
        return [r for r in rows if t0 <= r['start_sec'] and r['end_sec'] <= t1]

    residence = prefix(response['residence'])
    transfers = prefix(response['transfers'])
    resources = prefix(response['resource_allocations'])
    assert residence and min(r['start_sec'] for r in residence) == t0
    assert max(r['end_sec'] for r in residence) == t1
    costs, outside = {}, {}
    for row in residence:
        for key, n in row['inside_veh'].items():
            costs[key] = costs.get(key, 0.) + n*row['dt_h']
            extra = row['model_stock_veh'][key]-n
            assert extra >= -1e-7
            outside[key] = outside.get(key, 0.) + max(0., extra)*row['dt_h']

    for row in transfers:
        key = (row['source'], row['target'], row['route_key'])
        flows[key] = flows.get(key, 0.) + row['vehicles']
    ramps = {}
    for ramp in cfg.network.ramps:
        arrival = sum(r['vehicles'] for r in transfers if r['target']=='ramp:'+ramp)
        merge = sum(r['vehicles'] for r in transfers
                    if r['source']=='ramp:'+ramp and r['target']=='merge_pending:'+ramp)
        initial, final = state.ramp_queue[ramp], end.ramp_queue[ramp]
        assert abs(initial+arrival-merge-final) < 1e-7
        head = sum(r['accepted_total_veh'] for r in resources
                   if r['kind']=='physical_ramp_head_service' and r['resource'].startswith(ramp+':'))
        ramps[ramp] = dict(initial_stock=initial, arrival=arrival, head=head,
                           merge=merge, final_stock=final)
    offramps = {}
    for off, spec in state.lane_offramp_runtime.descriptions.items():
        key = 'storage:'+spec['storage']
        arrival = sum(r['vehicles'] for r in transfers if r['target']==key)
        drain = sum(r['vehicles'] for r in transfers if r['source']==key)
        initial = state.lane_offramp_runtime.stock(off)
        final = end.lane_offramp_runtime.stock(off)
        assert abs(initial+arrival-drain-final) < 1e-7, (off,initial,arrival,drain,final)
        offramps[off] = dict(storage=spec['storage'], connector=spec['connector'],
                            initial_stock=initial, arrival=arrival, drain=drain, final_stock=final)
    cells = []
    for s in (state, end):

                         cfg.network.freeway_segment_length_profile_km[road],s.freeway_effective_lanes[road])]
                  for road in s.freeway_density}
        assert all(len(v)==31 for v in counts.values())
        cells.append(dict(time_sec=s.time_sec, vehicle_count=counts,
                          speed_kmh=s.freeway_speed, density=s.freeway_density))
    return dict(start_sec=t0,end_sec=t1,future_observation_inputs=False,
                ttt_omega_veh_h=sum(costs.values()),cost_by_stock=costs,
                tracked_outside_veh_h=sum(outside.values()),ramps=ramps,offramps=offramps,
                physical_cell_states=cells,transfers=[dict(source=k[0],target=k[1],route_key=k[2],vehicles=v)
                    for k,v in flows.items()])


def executed_interval_responses(state, point, cfg):
    """Partition an already completed450s response; do not predict again."""
    from types import SimpleNamespace
    starts=(state,*point.states[:-1])
    rows=[first_interval_response(start,SimpleNamespace(states=[end],
             control_area_response=point.control_area_response),cfg)
          for start,end in zip(starts,point.states)]
    assert len(rows)==3 and abs(sum(row['ttt_omega_veh_h'] for row in rows)-point.ttt)<1e-7
    return rows

        report.update(purpose=('Autonomous450s fixed RM forecasts with additional unexecuted signal exchanges' if signal_exchange else
                              'Autonomous450s forecast of two verified fixed native RM sequences; no fitting or optimization'),
                      replay_pair=replay_pair)
        if replay_pair.get('seed') in (67,73):
            assert tuple(results) == ('release_actual','release_vsl90_actual')
            report.update(purpose='Autonomous450s VSL contrast under identical executed RM release schedules; no fitting or optimization',
                          reference_arm='release_actual',native_reference_arm='release',
                          prediction_conditioned_on_executed_commands=True)
    if four_arm_receipt is not None:
        assert len(results)==(6 if east_meter_progressive_screen else 5 if east_meter_first8_screen else 4 if candidate_names is None else len(candidate_names))
        assert all(hashlib.sha256(Path(p).read_bytes()).hexdigest()==h for p,h in profile_pins.items())
        report.update(purpose=f"Four executed seed{four_arm_seed} command profiles, current full coupled plant, no fit or optimization",
                      native_profile_inputs=profile_pins,reference_arm='held_actual',native_reference_arm='nc' if independent_four else 'none',
                      prediction_conditioned_on_executed_commands=True)
        if east_meter_first8_screen:
            report.update(purpose='Four first-block RM g10->g8 probes from verified seed43 NC state; model-only finite-step screen',
                          native_counterfactual=False,prediction_conditioned_on_executed_commands=False,
                          candidate_profile=[8,10,10],future_observation_inputs=False)
        if east_meter_progressive_screen:
            report.update(purpose='Four individual and one joint east RM8/6/4 sequences from verified seed43 NC state; model-only response screen',
                          native_counterfactual=False,prediction_conditioned_on_executed_commands=False,
                          candidate_profile=[8,6,4],future_observation_inputs=False)
    if source_document is not None:
        report.update(purpose='Future observed FW_E source ablation, conditional diagnosis only; NOT autonomous gain validation',
                      future_observation_inputs=True,diagnostic_source_contract=source_document,
                      autonomous_prediction=False)
    (output/'summary.json').write_text(json.dumps(report,ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')


def probe_selected_meter(captured, reference, coord, options, evaluate, selected_path, output, *, vsl_zone=None, components=False,
                         local_neighbors=False, execution_reference=None, east_flat8=False, ramp_key=None,
                         local_factorial=False, response_neighbors=False):
    """Three fixed candidates around a completed selection; never optimize or apply."""
    from evaluation.controllers import vissim_stackelberg_adapter as adapter
    from evaluation.controllers import sdmpc_sequence as sequence, sdmpc_pfo
    from evaluation.controllers import area_leader_objective as constraints
    from src.models.state import ControlAction
    from src.controllers import rollout_endpoint as endpoint
    cfg, state = captured['cfg'], captured['state']
    saved = json.loads(selected_path.with_suffix('.joint.json').read_bytes())['selection']
    binding = json.loads(selected_path.with_suffix('.joint_written.json').read_bytes())
    assert binding['written_command_binding_passed']
    assert binding['action_json']['sha256'] == hashlib.sha256(selected_path.read_bytes()).hexdigest()
    # Resolve the completed reference before spending any forecast budget.
    execution_path = (execution_reference or selected_path.parent.with_name(selected_path.parent.name.replace('_select_','_select_check_')))/'summary.json'
    old = json.loads(execution_path.read_bytes())
    if execution_reference is not None:
        assert Path(old['selected_path']).resolve()==selected_path.resolve()
        assert old['future_observation_inputs'] is False and old['optimizer_iterations']==0
    selected = adapter.control_from_json(selected_path, cfg, ControlAction)
    if sequence.KEY not in selected.diagnostics:
        selected = sequence.pack(sequence.actions(selected,3))
    coord.validate(selected)
    ramp = ramp_key or 'RM_C10484'
    assert ramp in cfg.network.ramps
    if ramp_key is not None:
        assert not (east_flat8 or local_neighbors or components or vsl_zone)
    original = sequence.actions(selected, 3)
    if east_flat8:
        assert state.time_sec in (2250,2700,4500) and not components and not local_neighbors and vsl_zone is None
        assert all(c.diagnostics['rw_meter_green_'+r] == 10. for c in original for r in cfg.network.ramps)
    elif local_neighbors:
        assert not components and vsl_zone is None and state.time_sec == 2700
        assert [c.vsl['FW_E__seg13'] for c in original] in ([100.,110.,110.],[110.,110.,110.])
        assert all(c.diagnostics['rw_meter_green_'+ramp] == 2. for c in original)
        assert execution_reference is not None
    elif components:
        assert vsl_zone is None
        assert all(c.vsl == reference.vsl for c in original)
    elif vsl_zone is None:
        assert reference.diagnostics['rw_meter_green_'+ramp] == 10
        assert all(c.diagnostics['rw_meter_green_'+ramp] == 10 for c in original)
    else:
        assert vsl_zone=='FW_E__seg13' and reference.vsl[vsl_zone]==110
        assert all(c.vsl[vsl_zone]==110 for c in original)
    candidates = {'selected': selected}
    if east_flat8:
        east_ramps = ('RM_C10639','RM_C10681','RM_C10490','RM_C10484')
        groups = {r+'_flat8': (r,) for r in east_ramps}
        groups['east_all_flat8'] = east_ramps
        for name, ramps in groups.items():
            z = coord.encode(selected)
            changed = []
            for j, axis in enumerate(coord.axes):
                if axis['kind'] != 'meter' or axis['key'] not in ramps:
                    continue
                assert axis['owner'] == 'FW_E' and 8. in axis['allowed']
                z[j] = (8.-reference.diagnostics['rw_meter_green_'+axis['key']])/axis['scale']
                changed.append((axis['block'],axis['key']))
            assert set(changed) == {(b,r) for b in range(3) for r in ramps}
            trial = coord.decode(z, selected)
            coord.validate(trial)
            for before, after in zip(original, sequence.actions(trial,3)):
                assert before.green_times == after.green_times and before.offsets == after.offsets
                assert before.vsl == after.vsl
                assert before.N_P_star == after.N_P_star and before.N_UF_star == after.N_UF_star
                for r in cfg.network.ramps:
                    assert after.diagnostics['rw_meter_green_'+r] == (8. if r in ramps else 10.)
                    if r not in ramps: assert before.ramp_metering[r] == after.ramp_metering[r]
            candidates[name] = trial
    if local_factorial:
        assert local_neighbors and execution_reference is not None
        assert all(c.vsl['FW_E__seg13'] == 110. for c in original)
        held_z, selected_z = coord.encode(reference), coord.encode(selected)
        held_blocks = sequence.actions(reference,3)
        affected = {'FW_E'} | {f'FW_E__seg{i}' for i,h in enumerate(
            cfg.network.freeway_vsl_zone_head_of_cell['FW_E']) if h == 13}
        for name, revert_rm, first_vsl in (
                ('city_only', True, False),
                ('selected_vsl100_first', False, True),
                ('city_only_vsl100_first', True, True)):
            z = selected_z.copy()
            vsl_changes = 0
            for j,axis in enumerate(coord.axes):
                if revert_rm and axis['kind'] == 'meter':
                    z[j] = held_z[j]
                if first_vsl and axis['kind'] == 'vsl' and axis['key'] == 'FW_E__seg13' and axis['block'] == 0:
                    assert 100. in axis['allowed']
                    z[j] = (100.-reference.vsl[axis['key']])/axis['scale']
                    vsl_changes += 1
            assert vsl_changes == int(first_vsl)
            trial = coord.decode(z,selected); coord.validate(trial)
            for block,(before,held,after) in enumerate(zip(original,held_blocks,sequence.actions(trial,3))):
                assert before.green_times == after.green_times and before.offsets == after.offsets
                assert before.N_P_star == after.N_P_star and before.N_UF_star == after.N_UF_star
                wanted = held if revert_rm else before
                assert after.ramp_metering == wanted.ramp_metering
                assert all(after.diagnostics['rw_meter_green_'+r] == wanted.diagnostics['rw_meter_green_'+r]
                           for r in cfg.network.ramps)
                if first_vsl and block == 0:
                    assert after.vsl['FW_E__seg13'] == 100.
                    assert all(before.vsl[k] == after.vsl[k] for k in before.vsl if k not in affected)
                else:
                    assert before.vsl == after.vsl
            candidates[name] = trial
    if response_neighbors:
        assert local_neighbors and not local_factorial and state.time_sec == 2700
        assert all(c.vsl['FW_E__seg13'] == 110. for c in original)
        assert all(c.diagnostics['rw_meter_green_RM_C10490'] == 5. for c in original)
        assert all(c.diagnostics['rw_meter_green_RM_C10681'] == 9. for c in original)
        profiles = {
            'rm10490_first3': {('meter','RM_C10490'):3.},
            'rm10490_first7': {('meter','RM_C10490'):7.},
            'rm10681_first7': {('meter','RM_C10681'):7.},
            'vsl100_first': {('vsl','FW_E__seg13'):100.},
            'rm10490_first3_vsl100_first': {('meter','RM_C10490'):3.,('vsl','FW_E__seg13'):100.},
        }
        for name,changes in profiles.items():
            z=coord.encode(selected); found=set()
            for j,axis in enumerate(coord.axes):
                key=(axis['kind'],axis['key'])
                if axis['block'] != 0 or key not in changes: continue
                value=changes[key]; assert value in axis['allowed']
                initial=reference.vsl[key[1]] if key[0]=='vsl' else reference.diagnostics['rw_meter_green_'+key[1]]
                z[j]=(value-initial)/axis['scale']; found.add(key)
            assert found==set(changes)
            trial=coord.decode(z,selected); coord.validate(trial)
            for block,(before,after) in enumerate(zip(original,sequence.actions(trial,3))):
                assert before.green_times==after.green_times and before.offsets==after.offsets
                assert before.N_P_star==after.N_P_star and before.N_UF_star==after.N_UF_star
                for r in cfg.network.ramps:
                    expected=changes.get(('meter',r),before.diagnostics['rw_meter_green_'+r]) if block==0 else before.diagnostics['rw_meter_green_'+r]
                    assert after.diagnostics['rw_meter_green_'+r]==expected
                    if block or ('meter',r) not in changes: assert after.ramp_metering[r]==before.ramp_metering[r]
                if block or ('vsl','FW_E__seg13') not in changes:
                    assert before.vsl==after.vsl
                else:
                    affected={'FW_E'}|{f'FW_E__seg{i}' for i,h in enumerate(cfg.network.freeway_vsl_zone_head_of_cell['FW_E']) if h==13}
                    assert after.vsl['FW_E__seg13']==100.
                    assert all(before.vsl[k]==after.vsl[k] for k in before.vsl if k not in affected)
            candidates[name]=trial
    if local_neighbors and not local_factorial and not response_neighbors:
        local_trials=([('vsl_held110','vsl','FW_E__seg13',110.)]
                      if original[0].vsl['FW_E__seg13']!=110. else [])
        local_trials += [('rm_first3','meter',ramp,3.),('rm_first4','meter',ramp,4.)]
        for name,kind,key,value in local_trials:
            z = coord.encode(selected)
            matches = [(j,a) for j,a in enumerate(coord.axes)
                       if a['block']==0 and a['kind']==kind and a['key']==key]
            assert len(matches)==1
            j,axis = matches[0]; assert value in axis['allowed']
            initial = reference.vsl[key] if kind=='vsl' else reference.diagnostics['rw_meter_green_'+key]
            z[j] = (value-initial)/axis['scale']
            trial = coord.decode(z,selected); coord.validate(trial)
            for block,(before,after) in enumerate(zip(original,sequence.actions(trial,3))):
                assert before.green_times==after.green_times and before.offsets==after.offsets
                if block or kind=='meter': assert before.vsl==after.vsl
                else:
                    affected={'FW_E'}|{f'FW_E__seg{i}' for i,h in enumerate(cfg.network.freeway_vsl_zone_head_of_cell['FW_E']) if h==13}
                    assert all(before.vsl[k]==after.vsl[k] for k in before.vsl if k not in affected)
                assert all(before.ramp_metering[r]==after.ramp_metering[r]
                           for r in cfg.network.ramps if block or kind=='vsl' or r!=ramp)
            candidates[name] = trial
    if components:
        # Mix actual selected and held coordinates; do not invent a new command.
        held_z, selected_z = coord.encode(reference), coord.encode(selected)
        held_blocks = sequence.actions(reference, 3)
        for name in ('city_only', 'rm_only'):
            z = selected_z.copy()
            for j, axis in enumerate(coord.axes):
                use_selected = (axis['kind'] == 'meter') == (name == 'rm_only')
                z[j] = selected_z[j] if use_selected else held_z[j]
            trial = coord.decode(z, selected)
            coord.validate(trial)
            for joint, held, action in zip(original, held_blocks, sequence.actions(trial, 3)):
                city, meter = (joint, held) if name == 'city_only' else (held, joint)
                assert action.green_times == city.green_times and action.offsets == city.offsets
                assert action.vsl == reference.vsl
                assert action.ramp_metering == meter.ramp_metering
                assert all(action.diagnostics['rw_meter_green_'+r] == meter.diagnostics['rw_meter_green_'+r]
                           for r in cfg.network.ramps)
            candidates[name] = trial
    for profile in (() if components or local_neighbors or east_flat8 else ('vsl100','vsl90') if vsl_zone else ('flat8', 'progressive')):
        z = coord.encode(selected)
        for j, axis in enumerate(coord.axes):
            if axis['kind'] != ('vsl' if vsl_zone else 'meter') or axis['key'] != (vsl_zone or ramp):
                continue
            value = float(profile.removeprefix('vsl')) if vsl_zone else 8 if profile == 'flat8' else 8-2*axis['block']
            assert value in axis['allowed']
            initial=reference.vsl[vsl_zone] if vsl_zone else reference.diagnostics['rw_meter_green_'+ramp]
            z[j] = (value-initial)/axis['scale']
        trial = coord.decode(z, selected)
        coord.validate(trial)
        for before, after in zip(original, sequence.actions(trial, 3)):
            assert before.green_times == after.green_times and before.offsets == after.offsets
            if vsl_zone:
                assert before.ramp_metering==after.ramp_metering
                affected={'FW_E'}|{f'FW_E__seg{i}' for i,h in enumerate(cfg.network.freeway_vsl_zone_head_of_cell['FW_E']) if h==13}
                assert all(before.vsl[k]==after.vsl[k] for k in before.vsl if k not in affected)
            else:
                assert before.vsl == after.vsl
                assert all(before.ramp_metering[r] == after.ramp_metering[r] for r in cfg.network.ramps if r != ramp)
        candidates[profile] = trial
    items = evaluate(list(candidates.values()), derivatives=False)
    follower = adapter._PHASE_VECTOR_FOLLOWER['ref']
    def budget(action, quantities):
        return constraints.shared_quantity_constraints(follower, action, quantities,
            start_sec=state.time_sec, horizon_steps=3,
            np_mode='cap', target_np_veh=saved['final_constraints']['np']['target'],
            np_tolerance_veh=saved['final_constraints']['np']['tolerance'],
            nuf_mode='cap', target_nuf_veh_h=saved['final_constraints']['nuf']['target'],
            nuf_tolerance_veh_h=saved['final_constraints']['nuf']['tolerance'])
    results = {}
    for (name, action), item in zip(candidates.items(), items):
        model = dict(ttt_omega_veh_h=item['objective_veh_h'],
            quantities=budget(action, item['quantities']),
            physical_model_feasible=sdmpc_pfo.model_feasible(item, options['shared_tolerance']))
        started = time.perf_counter()
        speed_terms=[]
        scope=(observe_first_interval_speed_terms(state.time_sec,speed_terms,horizon_sec=450,cells=range(16,26))
               if vsl_zone is not None else nullcontext())
        with scope:
            with sequence.prediction_scope(action, cfg, state.time_sec, 3) as visits:
                point = endpoint.evaluate_price_point(state, action, copy.deepcopy(captured['forecast'][:3]), (),
                    endpoint.ObjectiveSpec(cfg, depth_override=3, box_walk=False, score_mode='raw'), capture_response=True)
        assert visits == [0,1,2] and not point.aborted and len(point.states) == 3
        response = point.control_area_response
        physical_quantities = constraints.shared_urban_quantities(follower, response, start_sec=state.time_sec, horizon_steps=3)
        outside = sum(row['dt_h']*sum(max(0.,row['model_stock_veh'][key]-n) for key,n in row['inside_veh'].items()) for row in response['residence'])
        ramps = {}
        for r in cfg.network.ramps:
            arrivals = sum(x['vehicles'] for x in response['transfers'] if x['target']=='ramp:'+r)
            merges = sum(x['vehicles'] for x in response['transfers'] if x['source']=='ramp:'+r and x['target']=='merge_pending:'+r)
            final = point.states[-1].ramp_queue[r]
            residual = state.ramp_queue[r]+arrivals-merges-final
            assert abs(residual)<1e-7
            ramps[r] = dict(arrivals=arrivals,merges=merges,final_stock=final,residual=residual)
        execution = dict(ttt_omega_veh_h=point.ttt, tracked_outside_veh_h=outside,
            quantities=budget(action,physical_quantities), ramps=ramps,
            conditional_model_feasibility_witness=response.get('conditional_model_feasibility_witness'),
            model_constraint_coverage=response.get('model_constraint_coverage'),
            resource_max_exceedance=max((x['exceedance_veh'] for x in response['resource_allocations']),default=0.),
            wall_sec=time.perf_counter()-started)
        if response_neighbors or local_factorial or east_flat8 or (vsl_zone is None and not components and not local_neighbors):
            costs = {}
            for row in response['residence']:
                for key,n in row['inside_veh'].items():
                    costs[key] = costs.get(key,0.)+row['dt_h']*n
            assert abs(sum(costs.values())-point.ttt)<1e-7
            execution['cost_by_stock'] = costs
            execution['executed_control_blocks'] = visits
        if vsl_zone is not None:
            execution['executed_intervals']=executed_interval_responses(state,point,cfg)
            assert len(speed_terms)==450*10
            with gzip.open(output/(name+'.terms.gz'),'wt',encoding='utf-8') as stream:
                json.dump(speed_terms,stream,allow_nan=False)
        results[name] = dict(surrogate=model,execution=execution,commands=[
            dict(vsl=c.vsl,green_times=c.green_times,offsets=c.offsets,
                meters={r:c.diagnostics['rw_meter_green_'+r] for r in cfg.network.ramps}) for c in sequence.actions(action,3)])
        (output/(name+'.json')).write_text(json.dumps(results[name],ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')
        print(json.dumps(dict(case=name,surrogate=model['ttt_omega_veh_h'],execution=point.ttt,
                             quantity_feasible=execution['quantities']['feasible'])),flush=True)
    assert abs(results['selected']['surrogate']['ttt_omega_veh_h']-saved['selected_objective'])<1e-7
    assert abs(results['selected']['execution']['ttt_omega_veh_h']-old['results']['selected']['ttt_omega_veh_h'])<1e-7
    for name, result in results.items():
        for kind in ('surrogate','execution'):
            result[kind]['delta_ttt_omega_veh_h']=result[kind]['ttt_omega_veh_h']-results['selected'][kind]['ttt_omega_veh_h']
    report=dict(start_sec=state.time_sec,horizon_sec=450,results=results,
        selected_surrogate_and_execution_reproduced=True,selected_city_vsl_unchanged=vsl_zone is None and not components,
        selected_city_rm_unchanged=vsl_zone is not None,vsl_zone=vsl_zone,
        selected_budget_caps_frozen=True,surrogate_rollouts=len(results),execution_rollouts=len(results),
        optimizer_iterations=0,native_started=False,gain_qualified=False,
        scope=('FW_E__seg13 at100/90 only; selected city/RM sequence and current PFO caps fixed; no calibration/native'
               if vsl_zone else ramp+' only at the already selected city/VSL sequence; compare real actuator sequence and fixed current PFO caps, no calibration/native'))
    if local_neighbors:
        report.update(scope='First-block VSL110 orRM10484 GREEN3/4 around selected VSL100/GREEN2; later blocks, selected city and fixed caps unchanged',
                      first_block_neighbors=True,selected_city_vsl_unchanged=False,
                      future_observation_inputs=False,completed_selection_check_sha256=hashlib.sha256(execution_path.read_bytes()).hexdigest())
    if local_factorial:
        report.update(scope='Selected city sequence held fixed: actual previous versus selected RM, crossed with VSL110 or FW_E__seg13 VSL100 in block0 only. Later VSL blocks remain110; fixed NP/NUF caps; no optimization, refit or native.',
                      first_block_lever_factorial=True, selected_city_unchanged=True,
                      selected_city_rm_unchanged=False, selected_city_vsl_unchanged=False)
    if response_neighbors:
        report.update(scope='Six finite first-block actuator neighbors: selected,10490 GREEN3/7,10681 GREEN7,FW_E__seg13 VSL100,and10490 GREEN3+VSL100. Selected city/later commands/current PFO caps fixed; no optimization, refit or native.',
                      selected_city_unchanged=True,selected_city_rm_unchanged=False,
                      selected_city_vsl_unchanged=False,first_block_response_neighbors=True)
    if east_flat8:
        report.update(scope='Four individual east meters GREEN8/8/8 and all four together around the saved selection; selected city, VSL, west meters and NP/NUF caps fixed.',
                      future_observation_inputs=False,completed_selection_check_sha256=hashlib.sha256(execution_path.read_bytes()).hexdigest(),
                      targeted_ramps=list(east_ramps),native_execution_passed=False)
    if components:
        held = old['results']['held_actual']
        report['scope'] = 'Selected city-only and RM-only ablations; held/joint initial state, VSL and coefficients fixed. Predicted effects only, no native or fitting.'
        report['components'] = {}
        for kind, baseline in (('surrogate', saved['held_objective']), ('execution', held['ttt_omega_veh_h'])):
            costs = {k: v[kind]['ttt_omega_veh_h'] for k, v in results.items()}
            report['components'][kind] = dict(held_omega_veh_h=baseline,
                delta_city_only=costs['city_only']-baseline, delta_rm_only=costs['rm_only']-baseline,
                delta_joint=costs['selected']-baseline,
                incremental_rm_at_selected_city=costs['selected']-costs['city_only'],
                interaction=costs['selected']-costs['city_only']-costs['rm_only']+baseline)
        report['components']['execution']['held_tracked_outside_veh_h'] = held['tracked_outside_residence_veh_h']
        report['completed_selection_check_sha256'] = hashlib.sha256(
            (selected_path.parent.with_name(selected_path.parent.name.replace('_select_','_select_check_'))/'summary.json').read_bytes()).hexdigest()
    (output/'summary.json').write_text(json.dumps(report,ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')


def probe_meter_tail(captured, reference, output, *, ramp='RM_C10484', reference_dir=None):
    """Two fixed legal paths past450s; diagnostic only, no MPC horizon change."""
    from src.controllers import rollout_endpoint as endpoint
    from evaluation.controllers import physical_ramp_branches as branches
    from evaluation.controllers import sdmpc_sequence as sequence
    original_cfg,initial=captured['cfg'],captured['state'];reference=sequence.first_action(reference)
    assert original_cfg.mpc.horizon_steps==3 and original_cfg.simulation.T_c_sec==150
    # TimeBins collapse all arrivals beyond their declared end into one final
    # slot. Allocate the diagnostic horizon BEFORE the first rollout; never
    # extend or reuse a state whose future due times were already collapsed.
    for name in ('native_input_route_state','native_input_prehead_state'):
        assert 'aggregate' not in getattr(initial,name,{}), 'Need the original uncollapsed observation state'
    cfg=copy.deepcopy(original_cfg)
    cfg.mpc.horizon_steps=6
    spec=cfg.network.physical_ramp_branches['ramps'][ramp]
    reference_dir=reference_dir or HERE/'closedloop3000_meter2700_sequence_v1'
    paths={'held':[10]*6,'progressive':[8,6,4,2,2,2]}
    protocol=dict(ramp=ramp,paths=paths,initial_time=initial.time_sec,
        existing_mpc_horizon_sec=450,diagnostic_duration_sec=900,
        demand='Original three causal forecast intervals; repeat interval3 thereafter, no future observed inputs.',
        purpose='Does the legal green change sequence enter an effective metering range only after the current horizon?',
        caveat='Fixed-command sensitivity only; later demand is uncertain. No optimizer, NP/NUF candidate-feasibility or native-gain qualification.',
        service_by_green_vph=spec['service_by_green_veh_h'],new_native_runs=0,production_changed=False,

def replay_vsl_history(initialize, context, raw, state, cfg, output, *, derived_directory=None,
                       compare_online=True):
    """Rebuild a trial model's past cohorts without changing native evidence."""
    from types import SimpleNamespace as NS
    from evaluation.controllers import obs150_contract as oc
    from evaluation.controllers.lane_plant_runtime import bin_frame
    obs=raw[oc.RAW_STATE_KEY];cutoff=int(obs['sim_sec']);directory=Path(obs['directory'])
    history=output/'vsl_history';history.mkdir(exist_ok=False)
    pins={};checks=[]
    def read(path):
        data=path.read_bytes();pins[str(path)]=hashlib.sha256(data).hexdigest()
        return json.loads(data)
    readback=directory/'vsl_readback.csv';pins[str(readback)]=hashlib.sha256(readback.read_bytes()).hexdigest()
    for end in range(150,cutoff+1,150):
        source=read(directory/f'state_{end:06d}.json')
        observed=source[oc.RAW_STATE_KEY]
        assert observed['run_id']==obs['run_id'] and Path(observed['directory']).resolve()==directory.resolve()
        derived=read((derived_directory or directory/'obs150')/f'derived_{end:06d}.json')
        assert derived['window']==dict(start_s=end-150,end_s=end)
        assert derived['inputs']['raw_sha256']==oc.canonical_sha256(observed)
        if end==cutoff:
            target_state=state;inputs=raw
        else:
            frame=oc.load_frame(oc.resolve(observed,observed['frames']['current']['path']),
                                observed['frames']['current']['sha256'],end)
            target_state=NS(lane_freeway_runtime=state.lane_freeway_runtime,
                            freeway_density={},freeway_effective_lanes={})
            for road in state.lane_freeway_runtime.configs:
                cells,rows,dropped=bin_frame(context['geometry'],frame['vehicles'],road,drop_before_start=True)
                assert not dropped
                lanes=[cell['lane_km']/cell['length_km'] for cell in cells]
                target_state.freeway_effective_lanes[road]=lanes
                target_state.freeway_density[road]=[len(row)/(cell['length_km']*lane)
                                                   for row,cell,lane in zip(rows,cells,lanes)]
            inputs={oc.RAW_STATE_KEY:observed,oc.MERGED_DERIVED_KEY:derived}
        metadata=initialize(context,inputs,target_state,cfg,history_directory=history)
        actual=json.loads((history/f'vsl_cohorts_{end:06d}.json').read_bytes())
        if compare_online:
            expected=read(directory/'obs150'/f'vsl_cohorts_{end:06d}.json')
            for key in ('run_id','cutoff','frame_sha256','derived_sha256','applied_readbacks_sha256','cohorts','audit'):
                assert actual[key]==expected[key], ('Rebuilt physical VSL history differs',end,key)
        else:

            # recover that fact from actual readbacks and conserved past frames.
            assert all(float(cmd)==110. for rows in actual['cohorts'].values() for row in rows for cmd in row)
            assert all(abs(row['conservation_max'])<1e-7 for row in actual['audit'].values())
        checks.append(dict(cutoff=end,online_cache_compared=compare_online,
                           cohorts_exact=True if compare_online else None,audit_exact=True if compare_online else None))
    for path,digest in pins.items():assert hashlib.sha256(Path(path).read_bytes()).hexdigest()==digest
    receipt=dict(cutoff=cutoff,manifest_sha256=context['manifest_sha256'],checks=checks,
                 original_files_unchanged=True,original_files=pins,
                 scope='Measured past replay only; no model rollout, relabelled old cache or future observations')
    (output/'vsl_history_replay.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    return metadata

def verify_rm_replay_csvs(output, *, expected_files=44):
    """Run the actual VBS row validators over every prepared command, without VISSIM."""
    import re
    import subprocess
    source_path=ROOT/'scripts/run_real_world_stackelberg_controller.vbs'
    config_path=HERE/'selected/scenario/lane_native_b110.vbs'
    source=source_path.read_text(encoding='utf-8-sig')
    config=re.sub(r'&\s*_\r?\n\s*','&',config_path.read_text(encoding='utf-8-sig'))
    names=('ActionCsvHeaderValid','ExperimentMetadataValid','VslActionKeyValid','RampActionValid',
           'IsCanonicalNonNegativeInt','IsCanonicalCsvInt','IsFiniteNumberInRange',
           'IsCsvFiniteNumber','InCsvInt','InDelimitedText')
    functions=[re.search(r'(?ms)^Function '+name+r'\(.*?^End Function\s*$',source).group(0) for name in names]
    variables=('RW_ALLOWED_VSL_SPEEDS','RW_EXPECTED_VSL_DSD_IDS','RW_EXPECTED_VSL_ACTION_KEYS',
               'RW_RAMP_METER_IDS','RW_RAMP_METER_SCS','RW_RAMP_METER_CAPACITIES_VPH')
    assignments=[re.search(r'(?m)^'+name+r'\s*=.*$',config).group(0) for name in variables]
    cycle=re.search(r'(?m)^Const RAMP_CYCLE_SEC = .*$',source).group(0)
    files=sorted(output.glob('*/commands/action_*.csv')); assert len(files)==expected_files
    quoted=lambda p:'"'+str(p).replace('"','""')+'"'
    script='\n'.join(['Option Explicit',cycle,'Dim '+','.join(variables),
        'Dim RW_OFFSET_WRITER,fso,ts,p,line,parts,bad,rows,files',
        'RW_OFFSET_WRITER="experiment"',*assignments,
        'Set fso=CreateObject("Scripting.FileSystemObject")','bad=0:rows=0',
        'If ExperimentMetadataValid("") Then WScript.Quit 3',
        'files=Array('+','.join(map(quoted,files))+')',
        'For Each p In files', 'Set ts=fso.OpenTextFile(p,1,False)',
        'If Not ActionCsvHeaderValid(Split(ts.ReadLine,",")) Then bad=bad+1',
        'Do Until ts.AtEndOfStream','line=ts.ReadLine','parts=Split(line,",")','rows=rows+1',
        'If UBound(parts)<>14 Then','bad=bad+1','Else',
        'If Not ExperimentMetadataValid(parts(14)) Then bad=bad+1',
        'If parts(0)="vsl" Then',
        'If Not VslActionKeyValid(parts(1),parts(2),parts(4),parts(5)) Then bad=bad+1',
        'If Not IsCsvFiniteNumber(parts(6),RW_ALLOWED_VSL_SPEEDS) Then bad=bad+1',
        'ElseIf parts(0)="ramp_meter" Then',
        'If Not RampActionValid(parts(1),parts(3),parts(12),parts(13)) Then bad=bad+1',
        'Else','bad=bad+1','End If','End If','Loop','ts.Close','Next',
        'WScript.Echo CStr(rows) & "," & CStr(bad)','If bad>0 Then WScript.Quit 1',*functions])
    path=output/'csv_validation.vbs';path.write_text(script,encoding='utf-16')
    result=subprocess.run(['cscript.exe','//Nologo',str(path)],capture_output=True,text=True,encoding='mbcs',timeout=15)
    report=dict(returncode=result.returncode,stdout=result.stdout,stderr=result.stderr,files=len(files),
        source_pins={str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in (source_path,config_path,path)},
        empty_experiment_metadata_rejected=True,native_started=False)
    (output/'csv_validation.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    assert result.returncode==0 and result.stdout.strip()==f'{74*expected_files},0',report

        assert derived['window']==dict(start_s=sim_sec,end_s=sim_sec+150)
        assert derived['removals']['window_total']==len(derived['removals']['rows'])
        raw = captured['raw']
        for source in (raw,future):
            assert source['vehicle_records']['complete']
        with (HERE/'selected/obs150/obs150_detectors_v2.csv').open(encoding='utf-8-sig',newline='') as f:
            detectors = list(csv.DictReader(f))
        response = point.control_area_response
        if trace_mode:
            storage='SC1004_W_out'
            moves={m:v for m,v in cfg.network.urban_movements.items() if m.startswith('SC1004_')}
            def local_snapshot(s):
                return dict(movement_queue={m:s.urban_movement_queue.get(m,0.) for m in moves},
                    storage_free=s.urban_link_storage[storage],
                    pending=s.urban_storage_release_buffer.get(storage,{}),
                    known_routes=s.known_legsplit_route_state)
            trace=dict(movements=moves,initial=local_snapshot(state),final=local_snapshot(point.states[-1]),
                **({'gate_future_final_accounting':point.states[-1].gate_future_accounting}
                   if hasattr(point.states[-1],'gate_future_accounting') else {}),
                physical_initial_owners={p:a for p,a in state.local_observation_summary['projection_diagnostics']['physical_stock_assignment_by_link'].items()
                                        if a.get('storage:'+storage,0)>0},
                storage_capacity=cfg.network.urban_link_storage_veh[storage],
                greens={k:v for k,v in action.green_times.items() if 'SC1004' in k},
                offsets={k:v for k,v in action.offsets.items() if 'SC1004' in k},
                actual_head_window=[h for h in derived['head_window']['heads'] if h['sc']=='1004'],
                initial_model_observation_summary=state.local_observation_summary,
                response=response)
            if getattr(state.lane_ramp_runtime,'conflict_snapshot',{}):
                trace['ramp_conflict_snapshot']=state.lane_ramp_runtime.conflict_snapshot
            if hasattr(state,'gate_initial_route_evidence'):
                trace['gate_initial_route_evidence']=state.gate_initial_route_evidence
            with gzip.open(output/'trace.json.gz','wt',encoding='utf-8') as f:
                json.dump(trace,f,ensure_ascii=False,allow_nan=False)
        rows = {}
        for ramp in keys:
            link = int(ramp.removeprefix('RM_C'))
            def stock(source):
                return sum(int(v['link_no'])==link for v in source['vehicle_records']['records'])
            observed_start,observed_end = stock(raw),stock(future)
            assert abs(state.ramp_queue[ramp]-observed_start)<1e-7
            observed_arrival = derived['boundaries']['ramp_arrival:'+ramp]['cross']
            head_refs = {r['boundary_ref'] for r in detectors
                         if r['role']=='meter_head' and int(r['link'])==link}
            observed_head = sum(derived['boundaries'][ref]['cross'] for ref in head_refs)
            removed=sum(int(row['link'])==link for row in derived['removals']['rows'])
            observed_merge = observed_start+observed_arrival-observed_end-removed
            transfers = response['transfers']
            predicted_arrival = sum(r['vehicles'] for r in transfers if r['target']=='ramp:'+ramp)
            predicted_merge = sum(r['vehicles'] for r in transfers
                                  if r['source']=='ramp:'+ramp and r['target']=='merge_pending:'+ramp)
            predicted_head = sum(r['accepted_total_veh'] for r in response['resource_allocations']
                                 if r['kind']=='physical_ramp_head_service' and r['resource'].split(':')[0]==ramp)
            predicted_end = point.states[-1].ramp_queue[ramp]
            residual=observed_start+predicted_arrival-predicted_merge-predicted_end
            assert abs(residual)<1e-7, (ramp,residual)
            rows[ramp]=dict(initial_stock_veh=observed_start,
                actual=dict(arrival=observed_arrival,head=observed_head,merge=observed_merge,final_stock=observed_end),
                native_ramp_removals_veh=removed,
                predicted=dict(arrival=predicted_arrival,head=predicted_head,merge=predicted_merge,final_stock=predicted_end),
                stock_conservation_residual_veh=residual)
            sources={r['source'] for r in transfers if r['target']=='ramp:'+ramp}
            rows[ramp]['predicted_arrival_by_source']={str(source):sum(r['vehicles'] for r in transfers
                if r['target']=='ramp:'+ramp and r['source']==source) for source in sorted(sources)}
            rows[ramp]['prediction_minus_actual_veh']={k:rows[ramp]['predicted'][k]-v for k,v in rows[ramp]['actual'].items()}
        report.update(purpose='Recorded-command one-step response validation, not causal control-gain validation',
                      ramps=rows, future_target_only=True, removals=derived['removals'],
                      actual_merge_method='Initial connector stock + corrected entry crossings - final connector stock - native connector removals',
                      limitations=[f'Single recorded seed29 window{sim_sec}-{sim_sec+150}; no independent seed or prevention/recovery result.',
                                   'Recorded action includes urban green/offset and VSL changes; no lever-isolated inference.',
                                   'Native applied receipt checked; LDP/readback verification is reused from the prior run audit.'])
        report['target_inputs']={str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in (future_path,derived_path)}
        (output/'summary.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
        print(json.dumps(dict(summary=str(output/'summary.json'),ramps=rows)),flush=True)
        return
    baseline = points['original']
    for name in ('zero_proxy','hundredfold_proxy'):
        point = points[name]
        error = state_error(baseline.states,point.states)
        report['cases'][name]['all_stored_state_max_error'] = error
        report['cases'][name]['ttt_difference_veh_h'] = point.ttt-baseline.ttt
        report['cases'][name]['objective_difference'] = point.objective-baseline.objective
    report['proxy_has_no_physical_effect_in_this_probe'] = all(
        report['cases'][n]['all_stored_state_max_error']==0. and
        report['cases'][n]['objective_difference']==0.
        for n in ('zero_proxy','hundredfold_proxy'))
    report['limitations'] = [
        f'One recorded state at {sim_sec}s and one held 150s interval; not congestion/gain validation.',
        'Does not prove endogenous urban-to-ramp demand is correct.',
        'Leader proposal/initialization may still consume the unmasked proxy forecast.',
        'No parameter, demand configuration, physical model or controller search changed.']
    (output/'summary.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(dict(summary=str(output/'summary.json'),
                          proxy_invariant=report['proxy_has_no_physical_effect_in_this_probe'])),flush=True)
