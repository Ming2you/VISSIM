function Save-Post([string]$phase,[string]$detail){
 [ordered]@{phase=$phase;time=[DateTimeOffset]::Now.ToString('o');pid=$PID;creation=$taskCreated.ToString('o');native_owner_pid=14508;detail=$detail} |
  ConvertTo-Json -Depth 6 | Set-Content -LiteralPath $taskStatus -Encoding utf8

function Check-Stop {
 foreach($folder in @($runRoot,$reviewRoot,$protocolRoot)){

 }
}
function Verify-Sources {
 foreach($pin in $plan.source_pins.PSObject.Properties){
  if((Get-FileHash -LiteralPath $pin.Name -Algorithm SHA256).Hash.ToLowerInvariant() -cne $pin.Value){throw ('Postprocess source changed: '+$pin.Name)}
 }
 if((Get-FileHash -LiteralPath (Join-Path $reviewRoot 'candidate_config.json') -Algorithm SHA256).Hash.ToLowerInvariant() -cne $plan.config_sha256){throw 'Candidate configuration changed'}
 $original=Get-Content -Raw -LiteralPath (Join-Path $reviewRoot 'plan.json') | ConvertFrom-Json
 foreach($pin in $original.model_pins.PSObject.Properties){
  $expected=$pin.Value
  $override=$plan.source_pins.PSObject.Properties[$pin.Name]
  if($null -ne $override){$expected=$override.Value}
  if((Get-FileHash -LiteralPath $pin.Name -Algorithm SHA256).Hash.ToLowerInvariant() -cne $expected){throw ('Original pinned source changed: '+$pin.Name)}
 }
}
