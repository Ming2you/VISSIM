"""Measured head support joined to verified canonical physical resources.

No seed-max metadata is a measurement. No traffic or action is created here.
"""
from copy import deepcopy
import hashlib
import json
import math
from pathlib import Path
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[2]
EPS = 1e-8


def _read(pin):
    data = (ROOT / pin['path']).read_bytes()
    if hashlib.sha256(data).hexdigest() != pin['sha256']:
        raise ValueError('Physical head resource evidence changed')
    return data


def view(cfg):
    return getattr(cfg.network, 'head_service_resources', None)


def configure(cfg, tuning, raw, plan):
    section = (tuning or {}).get('urban', {}).get('capacity', {})
    path = section.get('head_resource_contract')
    if path is None:
        current = view(cfg)
        if current is not None:
            if current.get('schema') not in ('physical-head-resource-join/v1','physical-head-resource-join/v2','physical-head-resource-join/v3','physical-head-resource-join/v4','physical-head-resource-join/v5','physical-head-resource-join/v6','physical-head-resource-join/v7','physical-head-resource-join/v8','physical-head-resource-join/v9'):
                raise ValueError('Cannot clear an unowned physical head resource view')
            delattr(cfg.network, 'head_service_resources')
        return {}
    if not isinstance(path, str) or not path or section.get('head_observation', {}).get('enabled') is not True:
        raise ValueError('Physical resource contract requires enabled head observation')
    if tuning.get('urban', {}).get('shared_local_service_pool') is not True:
        raise ValueError('Physical head resource application requires the shared local service pool')
    data = (ROOT / path).read_bytes(); document = json.loads(data)
    catalogs={'physical-head-resource-join/v1':{'10619','10629'},
              'physical-head-resource-join/v2':{'10619','10629','10633'},
              'physical-head-resource-join/v3':{'10619','10629','10698'},
              'physical-head-resource-join/v4':{'10619','10629','10633','10698'},
              'physical-head-resource-join/v5':{'10619','10629','10633','10698','10283','10285'},
              'physical-head-resource-join/v6':{'10619','10629','10633','10698','10283','10285','10119'},
              'physical-head-resource-join/v7':{'10619','10629','10633','10698','10283','10285','10119','10691'},
              'physical-head-resource-join/v8':{'10619','10629','10633','10698','10283','10285','10119','10691','10539'},
              'physical-head-resource-join/v9':{'10619','10629','10633','10698','10283','10285','10119','10691','10539','10631'}}
    if set(document['resources']) != catalogs.get(document.get('schema')):
        raise ValueError('Unreviewed physical head resource set')
    network = _read(document['network'])
    if tuning.get('execution', {}).get('native_signal_record') is True:
        from evaluation.controllers.network_provenance import snapshot_physical_file_sha256
        matches = snapshot_physical_file_sha256(raw) == hashlib.sha256(network).hexdigest()
    else:
        matches = hashlib.sha256(Path(raw['network_path']).read_bytes()).digest() == hashlib.sha256(network).digest()
    if not matches:
        raise ValueError('Head resource and snapshot networks differ')
    from evaluation.controllers.signal_head_observation import physical_groups
    geometry = physical_groups(raw['network_path'], plan)
    tree = ET.fromstring(network); links = {n.get('no'): n for n in tree.findall('./links/link')}
    for connector, row in document['resources'].items():
        windows = row.get('green_exposure_windows', 0)
        if type(windows) is not int or windows not in (0, 4, 6, 8):
            raise ValueError('Invalid physical-resource green exposure pool')
        key = tuple(row['group'].split('|')); heads = geometry.get(key)
        if heads != row['heads'] or len({h['sg'] for h in heads}) != 1:
            raise ValueError('Physical resource head/lane/selected-SG binding changed')
        node = links[connector]; source = node.find('fromLinkEndPt'); target = node.find('toLinkEndPt')
        first = int(source.get('lane').split()[1]); lanes = len(node.findall('./lanes/lane'))
        if (source.get('lane').split()[0] != key[0] or target.get('lane').split()[0] != row['target_link']
                or set(range(first, first+lanes)) != {h['lane'] for h in heads}
                or any(h['position_m'] >= float(source.get('pos')) for h in heads)):
            raise ValueError('Head resource does not cover exactly its post-head connector lanes')
        # Every directly competing connector on these source lanes must branch
        # before the head. This rejects adding a post-head ambiguous resource.
        for other in links.values():
            edge = other.find('fromLinkEndPt')
            if edge is None or other.get('no') == connector or edge.get('lane').split()[0] != key[0]:
                continue
            lo = int(edge.get('lane').split()[1]); overlap = set(range(lo, lo+len(other.findall('./lanes/lane')))) & {h['lane'] for h in heads}
            if any(float(edge.get('pos')) >= h['position_m'] for h in heads if h['lane'] in overlap):
                raise ValueError('Another physical connector competes after the measured heads')
        proof = json.loads(_read(row['proof']))
        if row['mode'] == 'route_turn':
            incoming = [r for r in proof['incoming_turns'].values() if r['connector'] == connector]
            if len(incoming) != 1 or {incoming[0]['keep']} != set(row['members']) or incoming[0]['signal_controlled'] is not True:
                raise ValueError('Route-owned head resource lacks its exact finite-corridor turn')
        elif row['mode'] == 'regular_pool':
            for name in row['members']:
                evidence = [r for r in proof['topology_repairs'] if name in r['keep_movements']]
                if len(evidence) != 1 or evidence[0]['physical_stopline'] != key[0] or evidence[0]['keep_movements'][name] != '1123:2':
                    raise ValueError('Source alias lacks its verified common physical turn')
            decision = next(d for d in tree.findall('./vehicleRoutingDecisionsStatic/vehicleRoutingDecisionStatic') if d.get('no') == '1123')
            route = next(r for r in decision.findall('./vehRoutSta/vehicleRouteStatic') if r.get('no') == '2')
            if [n.get('key') for n in route.findall('./linkSeq/intObjectRef')] != [connector] or route.get('destLink') != row['target_link']:
                raise ValueError('Verified native resource route changed')
        elif row['mode'] in ('regular_unique','regular_exit'):
            if row['mode']=='regular_exit':
                # Crossing Omega does not remove the downstream receiving stock.
                # Verify this native exit and retain its existing outside queue.
                exits = {'10285': ('173','p3','SC109_W_SC108_to_E','SC108_to_SC109'),
                         '10283': ('1210009402','p2','SC109_N_SC16_to_E','SC16_to_SC109')}
                if connector not in exits:
                    raise ValueError('Unreviewed physical boundary exit')
                link,phase,name,origin = exits[connector]
                expected = {'kind':'boundary_out','signal':'SC109',
                            'phase':'SC109_'+phase,'origin':origin,'receiving_link':'SC109_E_out'}
                turn = proof.get('proposal', {}).get(name, {})
                from evaluation.controllers.control_area_objective import physical_membership_from_ledger
                membership = physical_membership_from_ledger(json.loads(_read(row['membership'])))
                actual = cfg.network.urban_movements.get(name,{})
                valid = (key==(link,phase) and row['target_link']=='1220000803'
                    and row['receiver']=='SC109_E_out' and proof.get('schema')=='movement_connector_map/v1'
                    and turn.get('connectors')==[connector] and turn.get('signal')=='SC109'
                    and turn.get('dest')=='OUT_LINK_'+row['target_link']
                    and membership.get(link) is True and membership.get(row['target_link']) is False
                    and not actual.get('ramp') and not actual.get('unsignalized')
                    and actual.get('receiving_link') in cfg.network.urban_link_storage_veh)
            elif connector in ('10691','10539'):
                link,phase,name,signal,origin,receiver,target_link,head_lanes = {
                    '10691':('329','p3','SC1002_E_SC101_to_W_SC1001','SC1002',
                             'SC101_to_SC1002','SC1002_to_SC1001','29',[2,3]),
                    '10539':('1220011503','p2','SC101_S_SC1_to_W_SC1002','SC101',
                             'SC1_to_SC101','SC101_to_SC1002','1220012402',[4,5]),
                }[connector]
                expected={'kind':'internal','signal':signal,'phase':signal+'_'+phase,
                          'origin':origin,'receiving_link':receiver}
                evidence=proof.get('movements',{}).get(name,{})
                authority=json.loads(_read(proof['authority']))['network']
                turns=authority['control_area_routes']['movement:'+name]['physical_turns']
                canonical=[t for t in turns if t.get('from_link')==link
                           and t.get('source_evidence',{}).get('canonical_approach_leg')]
                valid=(key==(link,phase) and proof.get('schema')=='head-queue-lane-support/v1'
                    and evidence.get('expected_spec')==expected
                    and evidence.get('from_link')==link and evidence.get('connector')==connector
                    and evidence.get('to_link')==row['target_link']==target_link and evidence.get('lanes')==head_lanes
                    and row['receiver']==expected['receiving_link'] and len(canonical)==1
                    and canonical[0].get('connector')==connector and canonical[0].get('to_link')==target_link
                    and all(authority['urban_movements'][name].get(k)==v for k,v in expected.items()))
            elif connector in ('10698','10119'):
                link,phase,name,origin,route_id = {
                    '10698':('40','p2','SC1001_S_SC1003_to_W_RAMP','SC1003_to_SC1001','1119:1'),
                    '10119':('29','p3','SC1001_E_SC1002_to_W_RAMP','SC1002_to_SC1001','1118:2'),
                }[connector]
                expected = {'kind':'boundary_out','signal':'SC1001','phase':'SC1001_'+phase,
                            'origin':origin,'receiving_link':'SC1001_W_out'}
                evidence = proof.get('evidence', {}).get(connector, {})
                valid = (key == (link,phase) and proof.get('schema') == 'physical-projection-support/v1'
                    and proof.get('link_to_storage', {}).get(connector) == row['receiver'] == expected['receiving_link']
                    and evidence.get('canonical_movements') == [name]
                    and evidence.get('physical_triplet') == [link,connector,'31']
                    and evidence.get('native_route_ids') == [route_id] and row['target_link'] == '31')
            elif connector=='10631':
                name='SC1004_S_to_N_SC1003'
                expected={'kind':'boundary_in','signal':'SC1004','phase':'SC1004_p1',
                          'origin':'in_SC1004_S','receiving_link':'SC1004_to_SC1003'}
                evidence=proof.get('sources',{}).get('in_SC1004_S',{})
                movement=evidence.get('movements',{}).get(name,{})
                valid=(key==('66','p1') and proof.get('schema')=='kinematic-urban-arrival/v1'
                    and evidence.get('link')=='66' and movement.get('connector')==connector
                    and row['target_link']=='47' and row['receiver']==expected['receiving_link']
                    and all(movement.get('expected_spec',{}).get(k)==v for k,v in expected.items()))
            else:
                name='SC1004_S_to_W'
                expected={'kind':'boundary_in','signal':'SC1004','phase':'SC1004_p2',
                          'origin':'in_SC1004_S','receiving_link':'SC1004_W_out'}
                valid = (connector=='10633' and key==('66','p2')
                    and proof.get('schema')=='known-wout-routes/v1'
                    and proof['incoming_movements'].get(name)==connector
                    and proof['storage']==row['receiver'] and row['target_link']=='68')
            if (not valid or row['members']!={name:expected}
                    or (row['mode']=='regular_exit' and any(
                        cfg.network.urban_movements.get(name,{}).get(k)!=v for k,v in expected.items()))
                    or (row['mode']!='regular_exit' and proof['network']['sha256']!=document['network']['sha256'])):
                raise ValueError('Unique regular head lacks its exact incoming route proof: '+connector)
            paths=[]
            for decision in tree.findall('./vehicleRoutingDecisionsStatic/vehicleRoutingDecisionStatic'):
                for route in decision.findall('./vehRoutSta/vehicleRouteStatic'):
                    path=[decision.get('link')]+[n.get('key') for n in route.findall('./linkSeq/intObjectRef')]+[route.get('destLink')]
                    if connector in path:paths.append(path)
            if not paths or any([key[0],connector,row['target_link']] not in [p[i:i+3] for i in range(len(p)-2)] for p in paths):
                raise ValueError('Unique regular service native path changed')
            competing={m for m,s in cfg.network.urban_movements.items()
                       if s.get('origin')==expected['origin'] and s.get('phase')==expected['phase'] and s.get('beta',0)>0}
            # A model phase may also name a right turn on a different lane,
            # or one that branches before the measured head on that lane.
            # Exclude it only with its own native path and destination mapping;
            # it never receives or shares this head's measured throughput.
            separate = row.get('disjoint_consumers', {})
            for other, detail in separate.items():
                actual = cfg.network.urban_movements.get(other, {})
                required = detail['expected_spec']
                path = detail['path']
                if (other == name or other not in competing or len(path) != 3 or path[0] != key[0]
                        or any(actual.get(k) != v for k,v in required.items())
                        or required.get('origin') != expected['origin'] or required.get('phase') != expected['phase']):
                    raise ValueError('Disjoint head consumer lacks its exact model source')
                turn = links[path[1]]; origin = turn.find('fromLinkEndPt'); end = turn.find('toLinkEndPt')
                lanes = set(range(int(origin.get('lane').split()[1]),
                                  int(origin.get('lane').split()[1])+len(turn.findall('./lanes/lane'))))
                mapping = json.loads(_read(detail['destination_mapping']))
                decision_no, route_no = detail['native_route'].split(':')
                decision = tree.find(f"./vehicleRoutingDecisionsStatic/vehicleRoutingDecisionStatic[@no='{decision_no}']")
                native = decision.find(f"./vehRoutSta/vehicleRouteStatic[@no='{route_no}']")
                route_path = [decision.get('link')]+[n.get('key') for n in native.findall('./linkSeq/intObjectRef')]+[native.get('destLink')]
                declared_path = detail.get('native_route_path', path)
                overlap=[h for h in heads if h['lane'] in lanes]
                upstream=(detail.get('upstream_of_head') is True and overlap
                          and all(float(origin.get('pos'))<h['position_m'] for h in overlap))
                if (origin.get('lane').split()[0] != key[0] or end.get('lane').split()[0] != path[2]
                        or not lanes or (overlap and not upstream)
                        or route_path != declared_path or declared_path[-3:] != path
                        or mapping.get('link_to_origins', {}).get(path[2]) != [required.get('receiving_link')]):
                    raise ValueError('Excluded consumer overlaps the head or lacks its native receiver')
                if 'downstream_choice' in detail:
                    # Before corridor installation the same pre-head turn has
                    # two destination aliases. Its later choice is not another
                    # consumer of this straight head's measured service.
                    choice_pin=detail['downstream_choice']; choice=json.loads(_read(choice_pin))
                    configured=tuning.get('urban',{}).get('route_choice_corridor',{}).get('evidence_paths',[])
                    turns=[t for t in choice.get('incoming_turns',{}).values()
                           if t.get('keep')==other and t.get('connector')==path[1]]
                    if (choice_pin['path'] not in configured or len(turns)!=1
                            or choice.get('network')!=document['network']
                            or turns[0].get('source_link')!=key[0]
                            or path[1] not in choice.get('prefix_links',[])
                            or path[2]!=choice.get('decision_link')):
                        raise ValueError('Pre-head destination alias lacks configured corridor proof')
                    alias=turns[0]['remove']; alias_spec=cfg.network.urban_movements.get(alias,{})
                    if (alias not in competing or alias==name
                            or any(alias_spec.get(k)!=required.get(k) for k in ('kind','signal','phase','origin'))
                            or alias_spec.get('receiving_link')!=choice.get('bypass_storage')):
                        raise ValueError('Pre-head destination alias changed its physical source')
                    competing.remove(alias)
            competing -= set(separate)
            if competing!={name}:
                raise ValueError('Unique regular head acquired another model consumer: '+connector+' '+repr(sorted(competing)))
        else:
            raise ValueError('Unsupported head resource consumer')
    cfg.network.head_service_resources = {**document, 'contract_sha256': hashlib.sha256(data).hexdigest(),
                                          'observations': {}, 'extra_pool_groups': {}}
    return {'physical_head_resource_contract_enabled': 1.}


def observe(original, cfg, raw, previous_path, caps, plan, distribute, options):
    resource = view(cfg)
    if resource is None:
        return original(cfg, raw, previous_path, caps, plan, distribute, options)
    # A verified one-head/one-turn join is independent of the legacy kind filter.
    # Do not widen that filter for other perimeter or pre-head bypass movements.
    unique={tuple(row['group'].split('|')):next(iter(row['members']))
            for row in resource['resources'].values() if row['mode'] in ('regular_unique','regular_exit')}
    legacy_distribute=distribute
    def physical_distribute(cfg_arg,groups,estimates,target):
        count=legacy_distribute(cfg_arg,groups,{k:v for k,v in estimates.items() if k not in unique},target)
        for key,name in unique.items():
            if key in estimates:
                target[name]=estimates[key];count+=1
        return count
    if unique:distribute=physical_distribute
    from evaluation.controllers.signal_head_observation import physical_groups, number, green_exposure_windows
    geometry = physical_groups(raw['network_path'], plan)
    groups = [{'stopline_link': k[0], 'signal': 'SC'+v[0]['sc']} for k, v in geometry.items()]
    memberships = {}
    for key in geometry:
        weights = {}; distribute(cfg, groups, {key: 1.}, weights); memberships[key] = set(weights)
    owned = {tuple(row['group'].split('|')) for row in resource['resources'].values()}
    managed = set().union(*(memberships[k] for k in owned))
    if any(managed & members for key, members in memberships.items() if key not in owned):
        raise ValueError('Managed resource members overlap another observer group')
    metadata = original(cfg, raw, previous_path, caps, plan, distribute, options)
    # Undo only the known provisional allocations. Canonical physical topology
    # still removes phantom movements, and later resource joining owns the rate.
    changed = dict(cfg.network.movement_capacity_by_movement_veh_h)
    for m in managed:
        changed[m] = caps[m]
    cfg.network.movement_capacity_by_movement_veh_h = changed
    doc = {}
    try:
        doc = json.loads(Path(previous_path).read_text(encoding='utf-8-sig'))
        prior = {**doc.get('diagnostics', {}), **doc.get('metadata', {})}
    except (OSError, TypeError, ValueError):
        prior = {}
    context = next(k for k in metadata if k.startswith('head_provenance_'))
    window = raw['local_observation'].get('signal_observation_window') or {}
    # The legacy observer does not recognize the new namespace as a prior.
    # Therefore independently verify the original action envelope even when
    # it contains only a carried observed-only value and no current candidates.
    try:
        previous_time = number(doc['metadata']['sim_sec'])
        previous_ok = (prior.get(context) == 1.0 and not metadata.get('head_observation_prior_discarded')
            and doc['run_provenance']['run_id'] == raw['run_provenance']['run_id']
            and number(prior['head_observation_snapshot_sec']) == previous_time == window.get('start_sec')
            and previous_time < number(raw['sim_sec'])
            and all(number(v) == previous_time for k,v in prior.items() if k.startswith('head_candidate_end_')))
    except (KeyError, TypeError, ValueError):
        previous_ok = False
    for connector, row in resource['resources'].items():
        suffix = hashlib.sha256((context+resource['contract_sha256']+connector).encode()).hexdigest()[:16]
        pool_windows=green_exposure_windows(cfg,tuple(row['group'].split('|')))
        # A disabled/different inference mode must not carry pooled estimates.
        if pool_windows:suffix+='_w'+str(pool_windows)
        name = 'head_resource_observed_floor_'+connector+'_'+suffix
        carried = number(prior.get(name, 0.)) if previous_ok else 0.
        if not math.isfinite(carried) or carried < 0:
            raise ValueError('Invalid previous observed-only physical resource support')
        prefix = row['group'].replace('|', '_')+'_'
        candidate = [k for k in metadata if k.startswith('head_candidate_rate_'+prefix)]
        support = 0.
        if candidate and previous_ok:
            rate_key = candidate[0]; end_key = rate_key.replace('head_candidate_rate_', 'head_candidate_end_')
            if prior.get(end_key) == window.get('start_sec') and metadata.get(end_key) == raw['sim_sec']:
                support = min(float(prior[rate_key]), float(metadata[rate_key]))
        pool_prefix='head_green_pool_'+prefix
        pool_floor=max((float(v) for k,v in metadata.items()
                        if k.startswith(pool_prefix) and k.endswith('_w'+str(pool_windows)+'_floor')),default=0.) if pool_windows else 0.
        observed = max(carried, support, pool_floor)
        metadata[name] = observed
        for key in tuple(metadata):
            if key.startswith('head_discharge_floor_'+prefix):
                metadata[key.replace('head_discharge_floor_', 'head_legacy_unapplied_floor_')] = metadata.pop(key)
        resource['observations'][connector] = {'observed_only_floor_veh_h': observed,
            'current_pair_support_veh_h': support, 'carried_observed_support_veh_h': carried,
            **({'pooled_observed_support_veh_h':pool_floor} if pool_windows else {}),
            'metadata_key': name, 'snapshot_sec': float(raw['sim_sec'])}
    metadata['head_resource_provisional_members_restored'] = float(len(managed))
    return metadata


def finalize(cfg):
    spec = view(cfg)
    if spec is None:
        return {}
    result = {}; caps = dict(cfg.network.movement_capacity_by_movement_veh_h)
    for connector, row in spec['resources'].items():
        if row['receiver'] not in cfg.network.urban_link_storage_veh:
            raise ValueError('Physical head resource has no finite receiving storage')
        for name, expected in row['members'].items():
            actual = cfg.network.urban_movements.get(name, {})
            if any(actual.get(k) != v for k, v in expected.items()) or actual.get('ramp'):
                raise ValueError('Final physical service member semantics changed')
        measured = spec['observations'][connector]['observed_only_floor_veh_h']
        if row['mode'] == 'route_turn':
            route = cfg.network.route_choice_corridor
            turns = route['turns']; name = next(iter(row['members'])); inherited = turns[name]['service_veh_h']
            if turns[name]['connector'] != connector or caps[name] != inherited:
                raise ValueError('Route physical service cap/turn mismatch before observation')
            selected = max(inherited, measured); caps[name] = selected
            for child in [route, *route.get('corridors', [])]:
                if name in child.get('turns', {}):
                    child['turns'][name]['service_veh_h'] = selected
        elif row['mode'] in ('regular_unique','regular_exit'):
            name=next(iter(row['members']))
            selected=max(caps[name],measured)
            caps[name]=selected
        else:
            # Comparison only, not a physical seed: preserve old behavior unless
            # observation supports an increase beyond the old aggregate envelope.
            envelope = sum(caps[m] for m in row['members'])
            if measured <= envelope + EPS:
                result['head_resource_unapplied_'+connector] = measured
                continue
            selected = measured
            for name in row['members']:
                caps[name] = selected
            first = next(iter(row['members'].values()))
            spec['extra_pool_groups'][connector] = {'members': tuple(row['members']),
                'signal': first['signal'], 'phase': first['phase'], 'receiver': row['receiver'],
                'service_veh_h': selected}
        result['head_resource_final_rate_'+connector] = selected
    cfg.network.movement_capacity_by_movement_veh_h = caps
    return result


def extend_local_pool(cfg):
    spec = view(cfg)
    if spec is None or not spec['extra_pool_groups']:
        return
    pool = getattr(cfg.network, 'local_service_pool', None)
    if not pool:
        raise ValueError('Observed shared head resource lacks its local pool')
    for key, row in spec['extra_pool_groups'].items():
        if key in pool['groups'] or any(m in pool['group_of'] for m in row['members']):
            raise ValueError('Physical resource is already registered in another pool')
        pool['groups'][key] = deepcopy(row)
        pool['group_of'].update({m: key for m in row['members']})


def regular_context(cfg, control, step):
    spec = view(cfg)
    groups = spec['extra_pool_groups'] if spec else {}
    from evaluation.controllers import local_signal_service as pool
    from src.models import urban_queue_model as uqm
    limits = {}; members = {}
    for key, row in groups.items():
        for name in row['members']:
            movement = cfg.network.urban_movements[name]
            if movement['kind'] != 'boundary_out' or movement.get('ramp'):
                raise ValueError('Regular resource acquired a non-regular service path')
            fraction = uqm._phase_green_fraction(control, cfg, movement, urban_step_index=step)
            pool.register_limit(limits, key, row['service_veh_h'], cfg.simulation.T_u_h, fraction)
            members[name] = key
    return limits, {}, members


def regular_batch(cfg, intended, context):
    limits, used, members = context
    if not members:
        return intended
    from evaluation.controllers.local_signal_service import limit_batch
    return limit_batch(intended, members, limits, used, cfg.urban_follower.receiving_space_rule)


def regular_accepted(movement, vehicles, context):
    limits, used, members = context
    if movement in members:
        from evaluation.controllers.local_signal_service import accepted
        accepted(vehicles, members[movement], limits, used)
